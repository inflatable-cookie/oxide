#!/usr/bin/env python3
"""Disposable oxide#082 submission recipe.

Starts the single original 20-minute session clock before any task-owned
setup effect, creates one fresh owned root under /private/tmp, binds the pin
and caps into packet.json and submits the one heavy-admitted Effigy run.
Not a product file and not a reusable launcher.
"""
import ctypes, hashlib, json, os, pathlib, shlex, shutil, subprocess, tempfile, time, uuid

STAGE = pathlib.Path(__file__).resolve().parent
PIN = '22d2f7709b9d3e9e826b0ce64f893292f8f6fda7'
SOURCE_REPOSITORY = '/Users/tom/Dev/projects/oxide'
CALLER = 'oxide:worker:082-final-output-links'
PYTHON = '/usr/bin/python3'

lib = ctypes.CDLL('/usr/lib/libSystem.B.dylib')
lib.mach_continuous_time.restype = ctypes.c_uint64
class Scale(ctypes.Structure):
    _fields_ = [('numer', ctypes.c_uint32), ('denom', ctypes.c_uint32)]
scale = Scale(); assert lib.mach_timebase_info(ctypes.byref(scale)) == 0

start = lib.mach_continuous_time() * scale.numer // scale.denom
utc_start = time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime())
pin = subprocess.run(['/usr/bin/git', '-C', SOURCE_REPOSITORY, 'rev-parse', '--verify', PIN + '^{commit}'],
                     capture_output=True, text=True, check=True).stdout.strip()
tree = subprocess.run(['/usr/bin/git', '-C', SOURCE_REPOSITORY, 'rev-parse', PIN + '^{tree}'],
                      capture_output=True, text=True, check=True).stdout.strip()
assert pin == PIN
v = os.statvfs('/private/tmp')
assert v.f_bavail * v.f_frsize >= 20 * 1024 ** 3, 'host below the 20 GiB reserve before effects'

root = pathlib.Path(tempfile.mkdtemp(prefix='oxide-082-links.', dir='/private/tmp')).resolve()
os.chmod(root, 0o700)
script = root / 'measure.py'
shutil.copyfile(STAGE / 'measure.py', script)
packet = {
    'root': str(root),
    'request_id': str(uuid.uuid4()),
    'caller': CALLER,
    'pin': PIN,
    'pin_tree': tree,
    'source_repository': SOURCE_REPOSITORY,
    'start_ns': start,
    'deadline_ns': start + 1200 * 1_000_000_000,
    'clock': 'raw mach_continuous_time via mach_timebase_info, started in submit.py before root creation, '
             'admission wait and setup',
    'utc_start': utc_start,
    'host_free_bytes_before_effects': v.f_bavail * v.f_frsize,
    'session_seconds_cap': 1200,
    'experimental_cargo_cap': 8,
    'setup_call_cap': 2,
    'image_size': '4g',
    'image_used_stop_bytes': int(3.5 * 1024 ** 3),
    'owned_scratch_cap_bytes': 6 * 1024 ** 3,
    'host_reserve_bytes': 20 * 1024 ** 3,
    'raw_log_cap_bytes': 32 * 1024 ** 2,
    'host_cargo_home': '/Users/tom/.cargo',
    'registry_use': 'read-only copy of Cargo.lock registry inputs into a scratch CARGO_HOME',
    'toolchain': '1.97.1-aarch64-apple-darwin',
    'toolchain_bin': '/Users/tom/.rustup/toolchains/1.97.1-aarch64-apple-darwin/bin',
    'python': PYTHON,
    'measure_sha256': hashlib.sha256(script.read_bytes()).hexdigest(),
    'submit_sha256': hashlib.sha256(pathlib.Path(__file__).read_bytes()).hexdigest(),
}
(root / 'packet.json').write_text(json.dumps(packet, indent=2) + '\n')
(root / 'effigy.toml').write_text(
    '[manifest]\nminimum_effigy_version = "0.14.0"\n'
    '[tasks.measurement]\nadmission = "heavy"\nrun = '
    + json.dumps('exec ' + shlex.join([PYTHON, '-I', str(script)])) + '\n')
print('SUBMIT root', root, 'request', packet['request_id'], flush=True)
os.chdir(root)
os.environ['EFFIGY_CALLER'] = CALLER
os.execvp('effigy', ['effigy', '--host-run-request-id', packet['request_id'], 'measurement'])
