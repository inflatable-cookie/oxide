#!/usr/bin/env python3
"""Disposable oxide#075 submission recipe.

Starts the single original 50-minute session clock before any task-owned
setup or install effect, creates one fresh owned root under /private/tmp,
binds the pin, caps and arm definitions into packet.json and submits the one
heavy-admitted Effigy run. Not a product file and not a reusable launcher.
"""
import ctypes, hashlib, json, os, pathlib, shlex, shutil, subprocess, sys, tempfile, time, uuid

STAGE = pathlib.Path(__file__).resolve().parent
PIN = 'dc84d039cd397c8292f37f4201e3916fa2ecd1d3'
SOURCE_REPOSITORY = '/Users/tom/Dev/projects/oxide'
CALLER = 'oxide:worker:075-cranelift-quick-run'

lib = ctypes.CDLL('/usr/lib/libSystem.B.dylib')
lib.mach_continuous_time.restype = ctypes.c_uint64
class Scale(ctypes.Structure):
    _fields_ = [('numer', ctypes.c_uint32), ('denom', ctypes.c_uint32)]
scale = Scale(); assert lib.mach_timebase_info(ctypes.byref(scale)) == 0

pin = subprocess.run(['/usr/bin/git', '-C', SOURCE_REPOSITORY, 'rev-parse', '--verify', PIN + '^{commit}'],
                     capture_output=True, text=True, check=True).stdout.strip()
assert pin == PIN

start = lib.mach_continuous_time() * scale.numer // scale.denom
root = pathlib.Path(tempfile.mkdtemp(prefix='oxide-075-cranelift.', dir='/private/tmp'))
os.chmod(root, 0o700)
(root / 'raw').mkdir()
script = root / 'measure.py'
shutil.copyfile(STAGE / 'measure.py', script)
toolchains = '/Users/tom/.rustup/toolchains/'
packet = {
    'root': str(root),
    'request_id': str(uuid.uuid4()),
    'caller': CALLER,
    'pin': PIN,
    'source_repository': SOURCE_REPOSITORY,
    'start_ns': start,
    'deadline_ns': start + 3000 * 1_000_000_000,
    'clock': 'raw mach_continuous_time via mach_timebase_info, started in submit.py before root creation, '
             'admission wait, install and setup',
    'utc_start': time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime()),
    'measured_cargo_cap': 29,
    'setup_cargo_cap': 2,
    'session_seconds_cap': 3000,
    'owned_scratch_cap_bytes': 8 * 1024 ** 3,
    'host_reserve_bytes': 20 * 1024 ** 3,
    'raw_log_cap_bytes': 32 * 1024 ** 2,
    'host_cargo_home': '/Users/tom/.cargo',
    'host_shell_note': 'the submitting shell exports CARGO_INCREMENTAL=0; every Cargo call uses an explicit '
                       'environment allowlist without it, so the dev-profile incremental default applies',
    'toolchains': {
        'stable': {'name': '1.97.1-aarch64-apple-darwin', 'bin': toolchains + '1.97.1-aarch64-apple-darwin/bin'},
        'nightly': {'name': 'nightly-2026-08-05-aarch64-apple-darwin',
                    'bin': toolchains + 'nightly-2026-08-05-aarch64-apple-darwin/bin'},
    },
    'arm_order': ['A', 'B', 'C', 'D'],
    'arms': {
        'A': {'toolchain': 'stable', 'what': 'stable 1.97.1, quick recipe, LLVM'},
        'B': {'toolchain': 'nightly', 'what': 'nightly-2026-08-05, quick recipe, LLVM (toolchain control)'},
        'C': {'toolchain': 'nightly', 'test': True,
              'what': 'nightly-2026-08-05, quick recipe, Cranelift for package oxide-storage only',
              'global_args': ['-Zcodegen-backend', '--config',
                              'profile.oxide-quick.package.oxide-storage.codegen-backend="cranelift"']},
        'D': {'toolchain': 'nightly', 'env': {'RUSTFLAGS': '-Zthreads=8'},
              'what': 'nightly-2026-08-05, quick recipe, LLVM, RUSTFLAGS=-Zthreads=8'},
    },
    'recipe_sha256': hashlib.sha256(script.read_bytes()).hexdigest(),
    'submit_sha256': hashlib.sha256(pathlib.Path(__file__).read_bytes()).hexdigest(),
}
(root / 'packet.json').write_text(json.dumps(packet, indent=2) + '\n')
(root / 'effigy.toml').write_text(
    '[manifest]\nminimum_effigy_version = "0.14.0"\n'
    '[tasks.measurement]\nadmission = "heavy"\nrun = '
    + json.dumps('exec ' + shlex.join([sys.executable, str(script)])) + '\n')
print('SUBMIT root', root, 'request', packet['request_id'], flush=True)
os.chdir(root)
os.environ['EFFIGY_CALLER'] = CALLER
os.execvp('effigy', ['effigy', '--host-run-request-id', packet['request_id'], 'measurement'])
