#!/usr/bin/env python3
"""Disposable oxide#080 submission recipe.

Starts the single original 40-minute session clock before any task-owned
setup effect, creates one fresh owned root under /private/tmp, binds the pin,
caps and arm definitions into packet.json and submits the one heavy-admitted
Effigy run. Not a product file and not a reusable launcher.
"""
import ctypes, hashlib, json, os, pathlib, shlex, shutil, subprocess, sys, tempfile, time, uuid

STAGE = pathlib.Path(__file__).resolve().parent
PIN = '669fd1e8ef4a6860de280ed0c0044d3415f30e02'
SOURCE_REPOSITORY = sys.argv[1]
CALLER = 'oxide:worker:080-stale-debug-objects'

lib = ctypes.CDLL('/usr/lib/libSystem.B.dylib')
lib.mach_continuous_time.restype = ctypes.c_uint64
class Scale(ctypes.Structure):
    _fields_ = [('numer', ctypes.c_uint32), ('denom', ctypes.c_uint32)]
scale = Scale(); assert lib.mach_timebase_info(ctypes.byref(scale)) == 0

start = lib.mach_continuous_time() * scale.numer // scale.denom
pin = subprocess.run(['/usr/bin/git', '-C', SOURCE_REPOSITORY, 'rev-parse', '--verify', PIN + '^{commit}'],
                     capture_output=True, text=True, check=True).stdout.strip()
assert pin == PIN
root = pathlib.Path(tempfile.mkdtemp(prefix='oxide-080-debugobj.', dir='/private/tmp'))
assert os.path.realpath(root) == str(root)
os.chmod(root, 0o700)
(root / 'raw').mkdir()
script = root / 'measure.py'
shutil.copyfile(STAGE / 'measure.py', script)
packet = {
    'root': str(root),
    'request_id': str(uuid.uuid4()),
    'caller': CALLER,
    'pin': PIN,
    'source_repository': SOURCE_REPOSITORY,
    'start_ns': start,
    'deadline_ns': start + 2400 * 1_000_000_000,
    'clock': 'raw mach_continuous_time via mach_timebase_info, started in submit.py before pin check, '
             'root creation, admission wait and setup',
    'utc_start': time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime()),
    'start_loadavg': os.getloadavg(),
    'measured_cargo_cap': 36,
    'setup_cargo_cap': 2,
    'session_seconds_cap': 2400,
    'owned_scratch_cap_bytes': 6 * 1024 ** 3,
    'host_reserve_bytes': 20 * 1024 ** 3,
    'raw_log_cap_bytes': 32 * 1024 ** 2,
    'host_cargo_home': '/Users/tom/.cargo',
    'host_shell_note': 'every Cargo call and executable uses an explicit environment allowlist; the '
                       'submitting shell environment (including any CARGO_INCREMENTAL) is not inherited',
    'toolchain': {'name': '1.97.1-aarch64-apple-darwin',
                  'bin': '/Users/tom/.rustup/toolchains/1.97.1-aarch64-apple-darwin/bin'},
    'arm_order': ['U', 'P', 'N'],
    'arms': {
        'U': {'what': 'quick recipe as-is: oxide-quick inherits dev, debug = "line-tables-only", default '
                      '(unpacked) split debug info'},
        'P': {'what': 'U plus split-debuginfo = "packed" for the oxide-quick profile',
              'env': {'CARGO_PROFILE_OXIDE_QUICK_SPLIT_DEBUGINFO': 'packed'}},
        'N': {'what': 'U plus debug = false for the single workspace package oxide-storage only, through a '
                      '--config package override; dependency units unchanged',
              'global_args': ['--config', 'profile.oxide-quick.package.oxide-storage.debug=false']},
    },
    'workload': 'per arm: cargo build --workspace --timings (cold) then ten edit -> cargo build --workspace '
                '--timings cycles, all --locked --offline --profile oxide-quick -v; then cargo test --no-run '
                '--test oxide_080_panic_probe and that executable run directly with RUST_BACKTRACE=1',
    'recipe_sha256': hashlib.sha256(script.read_bytes()).hexdigest(),
    'submit_sha256': hashlib.sha256(pathlib.Path(__file__).read_bytes()).hexdigest(),
}
(root / 'packet.json').write_text(json.dumps(packet, indent=2) + '\n')
(root / 'effigy.toml').write_text(
    '[manifest]\nminimum_effigy_version = "0.14.0"\n'
    '[tasks.measurement]\nadmission = "heavy"\nrun = '
    + json.dumps('exec ' + shlex.join([sys.executable, str(script)])) + '\n')
print('SUBMIT root', root, 'request', packet['request_id'], flush=True)
os.chdir(root)
os.environ['EFFIGY_CALLER'] = CALLER
os.execvp('effigy', ['effigy', '--host-run-request-id', packet['request_id'], 'measurement'])
