#!/usr/bin/env python3
"""Disposable oxide#080 continuation: same root, packet, clock and counters after
the attempt-1 harness defect (debug-map observer read CompletedProcess.pid).
Not a new session."""
import hashlib, json, os, pathlib, shutil, time, uuid
STAGE = pathlib.Path(__file__).resolve().parent
root = pathlib.Path('/private/tmp/oxide-080-debugobj.jkog1ebm')
packet = json.loads((root / 'packet.json').read_text())
assert not (root / 'measure-attempt-1.py').exists() and not (root / 'result-attempt-1.json').exists()
os.replace(root / 'measure.py', root / 'measure-attempt-1.py')
os.replace(root / 'result.json', root / 'result-attempt-1.json')
shutil.copyfile(STAGE / 'measure.py', root / 'measure.py')
prior = json.loads((root / 'result-attempt-1.json').read_text())
cont = {'attempt': 2, 'request_id': str(uuid.uuid4()), 'caller': packet['caller'],
        'utc_submitted': time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime()),
        'reason': 'attempt 1 stopped on AttributeError: the debug-map observer read .pid from a '
                  'subprocess.CompletedProcess. Both version calls and U-cold-build (exit 0) had run, and '
                  'the U cold census was recorded; the observer ran before any edit.',
        'attempt_1_counts': prior['counts'], 'attempt_1_stop_reason': prior['stop_reason'],
        'measure_sha256': hashlib.sha256((root / 'measure.py').read_bytes()).hexdigest(),
        'attempt_1_measure_sha256': hashlib.sha256((root / 'measure-attempt-1.py').read_bytes()).hexdigest(),
        'continue_sha256': hashlib.sha256(pathlib.Path(__file__).read_bytes()).hexdigest(),
        'unchanged': 'packet.json start_ns/deadline_ns, caps, root, counters, source checkout and target-U'}
assert cont['attempt_1_measure_sha256'] == packet['recipe_sha256']
(root / 'continuation.json').write_text(json.dumps(cont, indent=2) + '\n')
print('CONTINUE request', cont['request_id'], flush=True)
os.chdir(root); os.environ['EFFIGY_CALLER'] = packet['caller']
os.execvp('effigy', ['effigy', '--host-run-request-id', cont['request_id'], 'measurement'])
