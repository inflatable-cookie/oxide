#!/usr/bin/env python3
"""Derive summary.json from the retained oxide#080 result.json and
post-run-observation.json. No measurement."""
import json, statistics, sys, pathlib

root = pathlib.Path(sys.argv[1])
r = json.loads((root / 'result.json').read_text())
post = json.loads((root / 'post-run-observation.json').read_text())
MIB = 1024 ** 2
calls = {c['name']: c for c in r['calls']}
WS = ('lib', 'oxide-storage-lease "bin"', 'oxide-storage-inventory "bin"', 'oxide "bin"')
out = {'outcome': r['outcome'], 'stop_reason': r['stop_reason'], 'counts': r['counts'],
       'attempts': r.get('attempts'), 'pin': r['packet']['pin'], 'source': r['source'],
       'toolchain': r['packet']['toolchain']['name'], 'toolchain_identity': r['toolchain_identity'],
       'versions': {n: (root / calls[n]['stdout']).read_text() for n in ('cargo-version', 'rustc-version')},
       'utc_start': r['packet']['utc_start'], 'session': r['session'],
       'raw_log_bytes': r.get('raw_log_bytes'), 'boundaries': r['boundaries'],
       'owned_boundaries': r['storage'].get('owned_boundaries'),
       'peak_owned_allocated_mib': round(max(b['owned_allocated_bytes'] for b in r['storage']['owned_boundaries']) / MIB, 1),
       'min_host_free_gib': round(min(b['host_free'] for b in r['boundaries']) / 1024 ** 3, 1),
       'settings_comparison': r.get('settings_comparison'), 'post_run_observation': post['arms'],
       'arms': {}, 'notes': r['notes']}
for arm, a in r['arms'].items():
    cycles = []
    for b, c in zip(a['builds'], a['census']):
        d = next((x for x in a['debug_map'] if x['label'] == b['label']), {})
        units = calls[b['name']]['timings']['units']
        cycles.append({
            'label': b['label'], 'wall_s': b['wall_s'], 'exit': b['exit'], 'rebuilt_count': b['rebuilt_count'],
            'fresh_count': b['fresh_count'], 'loadavg_before': [round(x, 2) for x in b['loadavg_before']],
            'cargo_timing_unit_s': {u['target'].strip() or 'lib': u['duration'] for u in units
                                    if u['name'] == 'oxide-storage'},
            'deps_entries': c['deps_entries'], 'deps_files': c['deps_files'], 'rcgu_count': c['rcgu']['count'],
            'rcgu_apparent_mib': round(c['rcgu']['apparent'] / MIB, 2),
            'rcgu_allocated_sum_over_names_mib': round(c['rcgu']['allocated'] / MIB, 2),
            'rcgu_nlink_gt1': c['rcgu']['nlink_gt1'],
            'rcgu_same_inode_as_incremental': c['rcgu']['same_inode_as_incremental'],
            'target_allocated_unique_mib': round(c['target']['allocated_unique'] / MIB, 2),
            'target_apparent_mib': round(c['target']['apparent'] / MIB, 2),
            'by_area_allocated_unique_mib': {k: round(v['allocated_unique'] / MIB, 2) for k, v in c['by_area'].items()},
            'deps_dsym_dirs': len(c['deps_dsym_dirs']),
            'rcgu_referenced_by_current_binaries': d.get('deps_rcgu_referenced_by_current_binaries'),
            'rcgu_not_referenced_by_current_binaries': d.get('deps_rcgu_not_referenced_by_current_binaries')})
    warm = [x['wall_s'] for x in cycles if x['label'] != 'cold']
    after_panic = a['census'][-1]
    bt = a['backtrace']
    out['arms'][arm] = {
        'what': a['spec']['what'], 'resumed': a.get('resumed'), 'cycles': cycles,
        'cold_build_s': cycles[0]['wall_s'],
        'warm_build_s': {'mean': round(statistics.mean(warm), 4), 'median': round(statistics.median(warm), 4),
                         'min': min(warm), 'max': max(warm)},
        'warm_cargo_timing_unit_median_s': {k: statistics.median(x['cargo_timing_unit_s'][k] for x in cycles[1:])
                                           for k in WS},
        'rcgu_added_per_edit_cycle': sorted({cycles[i]['rcgu_count'] - cycles[i - 1]['rcgu_count']
                                             for i in range(1, len(cycles))}),
        'after_panic_build': {'deps_entries': after_panic['deps_entries'], 'rcgu_count': after_panic['rcgu']['count']},
        'panic_build': {k: a['panic_build'][k] for k in ('wall_s', 'exit', 'rebuilt')},
        'backtrace': {k: bt[k] for k in ('exit', 'expected_exit', 'panic_location', 'panic_location_matches',
                                         'frame_count', 'frames_with_file_line', 'probe_frames',
                                         'dsym_next_to_executable', 'frames')},
        'edits': [{k: e[k] for k in ('edit', 'file', 'old', 'new')} for e in a.get('edits', [])],
    }
(root / 'summary.json').write_text(json.dumps(out, indent=2) + '\n')
for arm, a in out['arms'].items():
    print(arm, a['cold_build_s'], a['warm_build_s'], a['warm_cargo_timing_unit_median_s'], a['rcgu_added_per_edit_cycle'],
          a['cycles'][-1]['rcgu_count'], a['backtrace']['frames_with_file_line'], a['backtrace']['frame_count'])
print(out['outcome'], out['counts'], out['session']['end_offset_s'], out['peak_owned_allocated_mib'], out['min_host_free_gib'])
