#!/usr/bin/env python3
"""Disposable oxide#080 stale debug-object hygiene measurement recipe.

Runs inside one heavy-admitted Effigy run. Not an Oxide product, not a reusable
launcher. One session on stable 1.97.1: two version calls, then three arms on
the quick recipe (U as-is: line tables, default unpacked split debug info;
P: + split-debuginfo = "packed"; N: workspace package debug = false through a
--config package override, dependencies unchanged). Per arm: one cold
`cargo build --workspace`, ten edit -> `cargo build --workspace` cycles on
src/dep_info.rs, then one `cargo test --no-run` of a scratch-only panicking
test, whose executable runs directly with RUST_BACKTRACE=1. After every build
the slot is observed read-only: deps/ entries, *.rcgu.o count, apparent and
unique allocated bytes, hardlinks and the current binaries' debug-map (OSO)
references. Nothing inside a target directory is ever deleted.
Caps: 36 measured Cargo calls + 2 version calls, the original 40-minute clock
started by the submission recipe, 6 GiB owned scratch, 20 GiB host reserve,
32 MiB raw logs. Never retries a call.
"""
import ctypes, errno, hashlib, json, os, pathlib, re, shlex, shutil, signal, subprocess, threading, time

ROOT = pathlib.Path(__file__).resolve().parent
RAW = ROOT / 'raw'
SRC = ROOT / 'src'
SCRATCH = ROOT / 't'
cfg = json.loads((ROOT / 'packet.json').read_text())
PIN = cfg['pin']
GIB = 1024 ** 3
PROFILE = 'oxide-quick'
WORKSPACE_PACKAGE = 'oxide-storage'

lib = ctypes.CDLL('/usr/lib/libSystem.B.dylib')
lib.mach_continuous_time.restype = ctypes.c_uint64
class Scale(ctypes.Structure):
    _fields_ = [('numer', ctypes.c_uint32), ('denom', ctypes.c_uint32)]
scale = Scale(); assert lib.mach_timebase_info(ctypes.byref(scale)) == 0
def now(): return lib.mach_continuous_time() * scale.numer // scale.denom
START = cfg['start_ns']; DEADLINE = cfg['deadline_ns']
def offset(): return round((now() - START) / 1e9, 3)

def free_bytes(path='/private/tmp'):
    v = os.statvfs(path); return v.f_bavail * v.f_frsize

class Stop(Exception):
    pass

counts = {'measured': 0, 'setup': 0}
result = {'packet': cfg, 'calls': [], 'arms': {}, 'storage': {}, 'boundaries': [], 'outcome': 'running',
          'stop_reason': None, 'counts': counts, 'notes': []}
scratch_reference = {'free': free_bytes()}

def save():
    tmp = ROOT / 'result.json.tmp'
    tmp.write_text(json.dumps(result, indent=2) + '\n'); tmp.replace(ROOT / 'result.json')

def raw_bytes():
    return sum(p.stat().st_size for p in RAW.iterdir() if p.is_file())

def bounds():
    if now() >= DEADLINE - 15_000_000_000:
        raise Stop('original 40-minute deadline reached (15 s margin)')
    free = free_bytes()
    if free < 21 * GIB:
        raise Stop('host free space %d below 20 GiB reserve + 1 GiB margin' % free)
    # Conservative owned-scratch guard while a call runs: any host free-space
    # drop since the reference counts as owned.
    if scratch_reference['free'] - free > int(5.5 * GIB):
        raise Stop('host free space dropped more than 5.5 GiB since the scratch reference')
    if raw_bytes() > 31 * 1024 ** 2:
        raise Stop('32 MiB raw log cap would be reached (1 MiB margin)')

def boundary(label):
    result['boundaries'].append({'label': label, 'offset_s': offset(), 'loadavg': os.getloadavg(),
                                 'host_free': free_bytes()})
    save()

def group_settled(pgid, limit=10.0):
    end = time.time() + limit
    while time.time() < end:
        try:
            os.killpg(pgid, 0)
        except OSError as error:
            if error.errno == errno.ESRCH:
                return 'ESRCH'
            if error.errno != errno.EPERM:
                return 'errno %d' % error.errno
        time.sleep(0.1)
    return None

def run(name, argv, env, kind, cwd=None, timeout=300, expect_zero=True):
    """kind: measured | setup | other. Counted calls increment before start."""
    bounds()
    if kind == 'measured':
        if counts['measured'] + 1 > cfg['measured_cargo_cap']:
            raise Stop('measured Cargo cap would be exceeded by ' + name)
        counts['measured'] += 1
    elif kind == 'setup':
        if counts['setup'] + 1 > cfg['setup_cargo_cap']:
            raise Stop('setup Cargo cap would be exceeded by ' + name)
        counts['setup'] += 1
    out = RAW / (name + '.stdout')
    err = RAW / (name + '.stderr')
    rec = {'name': name, 'kind': kind, 'argv': [str(a) for a in argv], 'cwd': str(cwd or ROOT),
           'env': env, 'start_offset_s': offset(), 'loadavg_before': os.getloadavg(),
           'free_before': free_bytes(), 'counts_after_start': dict(counts)}
    result['calls'].append(rec); save()
    print(name, 'START', rec['start_offset_s'], flush=True)
    t0 = now(); stop = None; exited = {}
    with out.open('wb') as o, err.open('wb') as e:
        child = subprocess.Popen([str(a) for a in argv], cwd=str(cwd or ROOT), env=env,
                                 stdin=subprocess.DEVNULL, stdout=o, stderr=e,
                                 start_new_session=True)
        rec['pgid'] = child.pid
        done = threading.Event()
        def wait_child():
            child.wait(); exited['ns'] = now(); done.set()
        threading.Thread(target=wait_child, daemon=True).start()
        try:
            while not done.wait(0.05):
                if (now() - t0) / 1e9 > timeout:
                    raise Stop('%s exceeded its %d s call bound' % (name, timeout))
                bounds()
        except Stop as reason:
            stop = reason
            try:
                os.killpg(child.pid, signal.SIGTERM)
                if not done.wait(5):
                    os.killpg(child.pid, signal.SIGKILL)
            except ProcessLookupError:
                pass
            done.wait()
    rec['wall_s'] = round((exited['ns'] - t0) / 1e9, 4)
    rec['exit'] = child.returncode
    rec['group_settled'] = group_settled(child.pid)
    rec['end_offset_s'] = offset(); rec['loadavg_after'] = os.getloadavg()
    rec['free_after'] = free_bytes()
    rec['stdout'] = str(out.relative_to(ROOT)); rec['stderr'] = str(err.relative_to(ROOT))
    save()
    print(name, 'EXIT', child.returncode, rec['wall_s'], flush=True)
    if stop is not None:
        rec['stopped'] = str(stop); save(); raise stop
    if rec['group_settled'] != 'ESRCH':
        raise Stop('%s process group not positively empty: %s' % (name, rec['group_settled']))
    if expect_zero and child.returncode != 0:
        raise Stop('%s exited %s; see retained raw logs' % (name, child.returncode))
    return rec

def sha256(path):
    h = hashlib.sha256()
    with open(path, 'rb') as f:
        for block in iter(lambda: f.read(1 << 20), b''):
            h.update(block)
    return h.hexdigest()

def du_k(path):
    out = subprocess.run(['/usr/bin/du', '-sk', str(path)], capture_output=True, text=True)
    return int(out.stdout.split()[0]) * 1024 if out.returncode == 0 else None

def owned_check(label):
    owned = du_k(ROOT)
    result['storage'].setdefault('owned_boundaries', []).append(
        {'label': label, 'offset_s': offset(), 'owned_allocated_bytes': owned, 'host_free': free_bytes(),
         'raw_log_bytes': raw_bytes()})
    save()
    if owned is None or owned > 6 * GIB:
        raise Stop('owned scratch %s exceeds 6 GiB or is unknown at %s' % (owned, label))

def crate_of(name):
    """Unit crate from a deps/ object name such as `oxide-<hash>.<cgu>.rcgu.o`."""
    return re.split(r'-[0-9a-f]{16}', name)[0]

def census(target):
    """Read-only slot observation; never modifies the target directory.

    Allocated bytes are st_blocks * 512 summed once per (dev, inode) across the
    whole target, so same-inode hardlinks are not double counted. APFS clone or
    snapshot sharing is invisible to st_blocks and is not accounted.
    """
    prof = target / PROFILE
    deps = prof / 'deps'
    seen = {}; links_in_tree = {}
    top = {'files': 0, 'dirs': 0, 'apparent': 0, 'allocated_unique': 0}
    by_area = {}
    for base, dirs, names in os.walk(target):
        dirs.sort()
        rel = pathlib.Path(base).relative_to(target).parts
        area = '/'.join(rel[:2]) if rel and rel[0] == PROFILE else (rel[0] if rel else '.')
        if rel and rel[0] == PROFILE and len(rel) >= 2 and rel[1] == 'deps' and len(rel) > 2:
            area = PROFILE + '/deps/<subdirectories>'
        top['dirs'] += len(dirs)
        for n in sorted(names):
            st = os.lstat(os.path.join(base, n))
            key = (st.st_dev, st.st_ino)
            links_in_tree[key] = links_in_tree.get(key, 0) + 1
            top['files'] += 1; top['apparent'] += st.st_size
            a = by_area.setdefault(area, {'files': 0, 'apparent': 0, 'allocated_unique': 0})
            a['files'] += 1; a['apparent'] += st.st_size
            if key in seen:
                continue
            seen[key] = st
            alloc = st.st_blocks * 512
            top['allocated_unique'] += alloc; a['allocated_unique'] += alloc
    inc_inodes = set()
    for base, _, names in os.walk(prof / 'incremental'):
        for n in names:
            st = os.lstat(os.path.join(base, n)); inc_inodes.add((st.st_dev, st.st_ino))
    entries = sorted(os.listdir(deps)) if deps.is_dir() else []
    rcgu = {'count': 0, 'apparent': 0, 'allocated': 0, 'nlink_gt1': 0, 'same_inode_as_incremental': 0,
            'by_crate': {}}
    names = []
    dep_files = 0; dsym_dirs = []
    for n in entries:
        p = deps / n; st = os.lstat(p)
        if p.is_dir():
            if n.endswith('.dSYM'):
                dsym_dirs.append(n)
            continue
        dep_files += 1
        if n.endswith('.rcgu.o'):
            names.append(n)
            rcgu['count'] += 1; rcgu['apparent'] += st.st_size; rcgu['allocated'] += st.st_blocks * 512
            rcgu['nlink_gt1'] += st.st_nlink > 1
            rcgu['same_inode_as_incremental'] += (st.st_dev, st.st_ino) in inc_inodes
            rcgu['by_crate'][crate_of(n)] = rcgu['by_crate'].get(crate_of(n), 0) + 1
    return {'offset_s': offset(), 'target': top, 'by_area': by_area,
            'deps_entries': len(entries), 'deps_files': dep_files, 'deps_dsym_dirs': dsym_dirs,
            'rcgu': rcgu, 'rcgu_names': names,
            'inodes_linked_twice_or_more_in_tree': sum(1 for v in links_in_tree.values() if v > 1)}

BINARIES = ('oxide', 'oxide-storage-inventory', 'oxide-storage-lease')

def debug_map(arm, label, target, census_rec, keep_lines):
    """Read the current uplifted binaries' OSO stabs (debug map) with nm.

    Records how many deps/*.rcgu.o objects the current binaries reference.
    A reference count is not a deletability proof: rlibs, incremental state and
    other artifacts are not examined.
    """
    prof = target / PROFILE
    referenced = set(); out = {}
    for b in BINARIES:
        path = prof / b
        if not path.exists():
            out[b] = {'missing': True}; continue
        bounds()
        t0 = now()
        p = subprocess.Popen(['/usr/bin/nm', '-ap', str(path)], stdout=subprocess.PIPE, stderr=subprocess.PIPE,
                             stdin=subprocess.DEVNULL, text=True, env={'PATH': '/usr/bin:/bin', 'HOME': '/Users/tom'},
                             start_new_session=True)
        stdout, _ = p.communicate(timeout=60)
        settled = group_settled(p.pid, 5.0)
        if settled != 'ESRCH':
            raise Stop('nm %s process group not positively empty: %s' % (b, settled))
        oso = [l for l in stdout.splitlines() if ' OSO ' in l]
        paths = [l.split(' OSO ', 1)[1].strip() for l in oso]
        kinds = {'deps_rcgu_o': 0, 'rlib_member': 0, 'other': 0}
        for q in paths:
            if '.rlib(' in q:
                kinds['rlib_member'] += 1
            elif q.startswith(str(prof / 'deps') + '/') and q.endswith('.rcgu.o'):
                kinds['deps_rcgu_o'] += 1; referenced.add(os.path.basename(q))
            else:
                kinds['other'] += 1
        existing = sum(1 for q in paths if '.rlib(' not in q and os.path.exists(q))
        out[b] = {'nm_exit': p.returncode, 'nm_group': settled, 'wall_s': round((now() - t0) / 1e9, 4),
                  'oso_count': len(paths), 'kinds': kinds, 'non_rlib_oso_paths_existing': existing,
                  'binary_sha256': sha256(path)}
        if keep_lines:
            (RAW / ('%s-%s-oso-%s.txt' % (arm, label, b))).write_text('\n'.join(oso) + '\n')
            out[b]['oso_lines'] = '%s-%s-oso-%s.txt' % (arm, label, b)
    names = set(census_rec['rcgu_names'])
    return {'binaries': out, 'deps_rcgu_referenced_by_current_binaries': len(names & referenced),
            'deps_rcgu_not_referenced_by_current_binaries': len(names - referenced),
            'referenced_but_missing': len(referenced - names)}

UNIT_DATA = re.compile(r'const UNIT_DATA = (\[.*?\]);\s*$', re.S | re.M)

def timing_units(arm, label, rec):
    """Copy Cargo's own --timings report and keep its per-unit durations."""
    stderr = (ROOT / rec['stderr']).read_text(errors='replace')
    m = re.search(r'Timing report saved to (\S+\.html)', stderr)
    if not m:
        rec['timings'] = {'found': False}; save(); return
    src = pathlib.Path(m.group(1))
    dst = RAW / ('%s-%s-cargo-timing.html' % (arm, label))
    shutil.copyfile(src, dst)
    text = dst.read_text(errors='replace')
    data = UNIT_DATA.search(text)
    units = None
    if data:
        try:
            units = [{k: u.get(k) for k in ('name', 'version', 'mode', 'target', 'start', 'duration',
                                            'rmeta_time')} for u in json.loads(data.group(1))]
        except ValueError as error:
            units = 'unparsed: %r' % (error,)
    rec['timings'] = {'found': True, 'report': str(src), 'copy': str(dst.relative_to(ROOT)),
                      'units': units}
    save()

def rustc_unit(line):
    """Normalise one verbose rustc line: debug settings apart, all other settings."""
    cmd = line.strip()[len('Running `'):]
    cmd = cmd[:-1] if cmd.endswith('`') else cmd
    tokens = shlex.split(cmd)
    crate = tokens[tokens.index('--crate-name') + 1]
    source = next((t for t in tokens if t.endswith('.rs')), None)
    registry = cfg['host_cargo_home'] + '/registry/src/'
    workspace = source is not None and not source.startswith('/')
    debug = {}; rest = []; i = 1
    while i < len(tokens):
        t = tokens[i]
        if t in ('--out-dir', '-L', '--extern'):
            i += 2; continue
        if t == '-C' and i + 1 < len(tokens):
            v = tokens[i + 1]
            k = v.split('=', 1)[0]
            if k in ('debuginfo', 'split-debuginfo', 'strip'):
                debug[k] = v.split('=', 1)[1] if '=' in v else ''; i += 2; continue
            if k in ('metadata', 'extra-filename', 'incremental'):
                rest.append('-C ' + k + '=<varies>'); i += 2; continue
            rest.append('-C ' + v); i += 2; continue
        rest.append(re.sub(r'target-[A-Z]\b', 'target-<arm>',
                           t.replace(registry, '<registry>/').replace(str(ROOT), '<root>')))
        i += 1
    kinds = tokens[tokens.index('--crate-type') + 1] if '--crate-type' in tokens else ('test' if '--test' in tokens else '?')
    if '--test' in tokens:
        kinds = 'test'
    key = '%s|%s|%s' % (crate, kinds, (source or '').replace(registry, '<registry>/'))
    return {'key': key, 'crate': crate, 'workspace': workspace, 'debug_flags': debug,
            'incremental': '-C incremental=' in cmd, 'build_script': (source or '').endswith('build.rs'),
            'settings_except_debug_sha256': hashlib.sha256(' '.join(rest).encode()).hexdigest()}

def parse_cargo(rec):
    stdout = (ROOT / rec['stdout']).read_text(errors='replace')
    stderr = (ROOT / rec['stderr']).read_text(errors='replace')
    artifacts = []; finished = None; executables = []
    for line in stdout.splitlines():
        if not line.startswith('{'):
            continue
        try:
            m = json.loads(line)
        except ValueError:
            continue
        if m.get('reason') == 'compiler-artifact':
            profile = m.get('profile') or {}
            artifacts.append({'package_id': m['package_id'], 'target': m['target']['name'],
                              'kinds': m['target']['kind'], 'fresh': m['fresh'], 'profile': profile,
                              'executable': m.get('executable')})
            if profile.get('test') is True and m.get('executable'):
                executables.append({'target': m['target']['name'], 'executable': m['executable']})
        elif m.get('reason') == 'build-finished':
            finished = m.get('success')
    units = [rustc_unit(l) for l in stderr.splitlines()
             if l.strip().startswith('Running `') and ' --crate-name ' in l]
    rec['finished_line'] = [l.strip() for l in stderr.splitlines() if l.strip().startswith('Finished ')]
    rec['build_finished_success'] = finished
    rec['rebuilt'] = sorted({a['target'] + ':' + ','.join(a['kinds']) + ':' + str(a['profile'].get('test'))
                             for a in artifacts if not a['fresh']})
    rec['fresh_count'] = sum(1 for a in artifacts if a['fresh'])
    rec['rebuilt_count'] = sum(1 for a in artifacts if not a['fresh'])
    rec['artifact_profiles'] = [{'package_id': a['package_id'].split('#')[-1],
                                 'workspace_member': a['package_id'].startswith('path+file://'),
                                 'target': a['target'], 'kinds': a['kinds'], 'fresh': a['fresh'],
                                 'debuginfo': a['profile'].get('debuginfo'), 'test': a['profile'].get('test')}
                                for a in artifacts]
    rec['rustc_units'] = units
    rec['test_executables'] = executables
    save()
    return executables

EDIT_FILE = SRC / 'src/dep_info.rs'
EDIT_REL = 'src/dep_info.rs'
BASE_VALUE = 32768
EDIT_LINE = 'pub const MAX_REFERENCES: usize = %d;\n'
PROBE_REL = 'tests/oxide_080_panic_probe.rs'
PROBE_NAME = 'oxide_080_panic_probe'
PROBE_SOURCE = '''//! Scratch-only oxide#080 probe: one uncaught panic through two helpers.

#[inline(never)]
fn probe_panic_site() -> ! {
    panic!("oxide-080-panic-probe");
}

#[inline(never)]
fn probe_caller() -> ! {
    probe_panic_site();
}

#[test]
fn oxide_080_deliberate_panic() {
    probe_caller();
}
'''

def probe_expected():
    lines = PROBE_SOURCE.splitlines()
    def line_of(fragment):
        return next(i + 1 for i, l in enumerate(lines) if fragment in l)
    return {'panic_location': '%s:%d:5' % (PROBE_REL, line_of('panic!(')),
            'frames': {'probe_panic_site': line_of('panic!('), 'probe_caller': line_of('probe_panic_site();'),
                       'oxide_080_deliberate_panic': line_of('    probe_caller();')}}

def edit(n):
    text = EDIT_FILE.read_text()
    old = EDIT_LINE % (BASE_VALUE + n - 1); new = EDIT_LINE % (BASE_VALUE + n)
    assert text.count(old) == 1, 'edit %d anchor not unique' % n
    EDIT_FILE.write_text(text.replace(old, new))
    return {'edit': n, 'file': EDIT_REL, 'old': old.rstrip('\n'), 'new': new.rstrip('\n'),
            'sha256_after': sha256(EDIT_FILE), 'offset_s': offset()}

def git(*args, name):
    return run(name, ['/usr/bin/git', '-C', SRC] + list(args), {'PATH': '/usr/bin:/bin', 'HOME': '/Users/tom'},
               'other', timeout=60)

FRAME = re.compile(r'^\s*(\d+): (.+)$')
AT = re.compile(r'^\s+at (.+?):(\d+)(?::(\d+))?$')

def analyse_backtrace(text):
    exp = probe_expected()
    loc = re.search(r"panicked at (\S+?):(\d+):(\d+):", text)
    frames = []; in_bt = False
    for line in text.splitlines():
        if line.startswith('stack backtrace:'):
            in_bt = True; continue
        if not in_bt:
            continue
        m = FRAME.match(line)
        if m:
            frames.append({'index': int(m.group(1)), 'symbol': m.group(2).strip(), 'at': None}); continue
        a = AT.match(line)
        if a and frames and frames[-1]['at'] is None:
            frames[-1]['at'] = {'file': a.group(1), 'line': int(a.group(2))}; continue
        if line.strip().startswith('note:') or not line.strip():
            in_bt = False
    probe = {}
    for fn, want in exp['frames'].items():
        f = next((f for f in frames if f['symbol'].endswith('::' + fn) or f['symbol'] == fn), None)
        probe[fn] = {'frame_present': f is not None,
                     'has_file_line': bool(f and f['at']),
                     'file_line': ('%s:%d' % (f['at']['file'], f['at']['line'])) if f and f['at'] else None,
                     'matches_expected_line': bool(f and f['at'] and f['at']['file'].endswith(PROBE_REL)
                                                   and f['at']['line'] == want)}
    return {'expected': exp,
            'panic_location': ('%s:%s:%s' % loc.groups()) if loc else None,
            'panic_location_matches': bool(loc and '%s:%s:%s' % loc.groups() == exp['panic_location']),
            'frame_count': len(frames), 'frames_with_file_line': sum(1 for f in frames if f['at']),
            'probe_frames': probe, 'frames': frames}

def measure_arm(arm, resume=False):
    spec = cfg['arms'][arm]
    target = ROOT / ('target-' + arm)
    env = base_env(); env.update(spec.get('env', {})); env['CARGO_TARGET_DIR'] = str(target)
    pre = [cfg['toolchain']['bin'] + '/cargo'] + spec.get('global_args', [])
    common = ['--locked', '--offline', '--profile', PROFILE, '-v', '--message-format=json-render-diagnostics']
    if resume:
        # Attempt 1 completed this arm's cold build and cold census, then stopped
        # in the debug-map observer. Continue on the same source and target.
        arm_rec = result['arms'][arm]
        assert [b['label'] for b in arm_rec['builds']] == ['cold'] and arm_rec['builds'][0]['exit'] == 0
        assert [c['label'] for c in arm_rec['census']] == ['cold'] and not arm_rec['debug_map']
        assert env == arm_rec['env'], 'resumed arm environment differs'
        boundary(arm + ' resume')
        head = git('rev-parse', 'HEAD', name=arm + '-resume-rev-parse')
        assert (ROOT / head['stdout']).read_text().split() == [PIN], 'resumed source HEAD is not the pin'
        status = git('status', '--porcelain', name=arm + '-resume-status')
        assert (ROOT / status['stdout']).read_text() == '?? %s\n' % PROBE_REL, 'resumed source not as left'
        assert sha256(SRC / PROBE_REL) == arm_rec['probe_sha256'], 'probe changed'
        assert sha256(EDIT_FILE) == arm_rec['baseline_edit_file_sha256'], 'edit file changed'
        arm_rec['resumed'] = {'offset_s': offset(), 'source_verified': True,
                              'cold_census_offset_s': arm_rec['census'][0]['offset_s']}
        save()
    else:
        arm_rec = {'spec': spec, 'env': env, 'builds': [], 'census': [], 'debug_map': []}
        result['arms'][arm] = arm_rec
        boundary(arm + ' start')
        git('reset', '--quiet', '--hard', PIN, name=arm + '-reset')
        git('clean', '-fdq', name=arm + '-clean')
        status = git('status', '--porcelain', name=arm + '-status')
        assert (ROOT / status['stdout']).read_text() == '', 'source not clean after reset'
        (SRC / PROBE_REL).write_text(PROBE_SOURCE)
        arm_rec['probe_sha256'] = sha256(SRC / PROBE_REL)
        arm_rec['baseline_edit_file_sha256'] = sha256(EDIT_FILE)
        save()
    for n in range(0, 11):
        label = 'cold' if n == 0 else 'e%d' % n
        if n == 0 and resume:
            c = arm_rec['census'][0]
            again = census(target)
            arm_rec['resumed']['cold_recensus'] = {
                'offset_s': again['offset_s'],
                'matches_attempt_1': all(again[k] == c[k] for k in ('target', 'by_area', 'deps_entries',
                                                                    'rcgu', 'rcgu_names'))}
            save()
            if not arm_rec['resumed']['cold_recensus']['matches_attempt_1']:
                raise Stop('%s target changed between attempt 1 and resume' % arm)
            d = debug_map(arm, label, target, c, keep_lines=True); d['label'] = label
            d['note'] = 'taken on resume, after a read-only re-census matched the attempt-1 cold census'
            arm_rec['debug_map'].append(d); save()
            owned_check(arm + ' after cold')
            boundary('%s after %s' % (arm, label))
            continue
        if n:
            arm_rec.setdefault('edits', []).append(edit(n)); save()
        rec = run('%s-%s-build' % (arm, label), pre + ['build', '--workspace', '--timings'] + common, env,
                  'measured', cwd=SRC, timeout=600 if n == 0 else 300, expect_zero=False)
        parse_cargo(rec)
        timing_units(arm, label, rec)
        arm_rec['builds'].append({'label': label, 'name': rec['name'], 'wall_s': rec['wall_s'],
                                  'exit': rec['exit'], 'fresh_count': rec['fresh_count'],
                                  'rebuilt_count': rec['rebuilt_count'], 'rebuilt': rec['rebuilt'],
                                  'finished_line': rec['finished_line'],
                                  'loadavg_before': rec['loadavg_before'], 'loadavg_after': rec['loadavg_after']})
        save()
        if rec['exit'] != 0:
            raise Stop('%s exited %s' % (rec['name'], rec['exit']))
        c = census(target); c['label'] = label
        arm_rec['census'].append(c); save()
        d = debug_map(arm, label, target, c, keep_lines=label in ('cold', 'e10')); d['label'] = label
        arm_rec['debug_map'].append(d); save()
        if n == 0:
            check_arm_control(arm, rec)
            owned_check(arm + ' after cold')
        boundary('%s after %s' % (arm, label))
    owned_check(arm + ' after cycles')
    rec = run('%s-panic-build' % arm, pre + ['test', '--no-run', '--test', PROBE_NAME] + common, env,
              'measured', cwd=SRC, timeout=300, expect_zero=False)
    executables = parse_cargo(rec)
    arm_rec['panic_build'] = {'name': rec['name'], 'wall_s': rec['wall_s'], 'exit': rec['exit'],
                              'rebuilt': rec['rebuilt'], 'executables': executables}
    save()
    if rec['exit'] != 0 or len(executables) != 1:
        raise Stop('%s exited %s with %d test executables' % (rec['name'], rec['exit'], len(executables)))
    c = census(target); c['label'] = 'after-panic-build'
    arm_rec['census'].append(c); save()
    tenv = {'PATH': '/usr/bin:/bin', 'HOME': '/Users/tom', 'LANG': 'C', 'RUST_BACKTRACE': '1',
            'OXIDE_TASK_SCRATCH_ROOT': str(SCRATCH), 'TMPDIR': str(SCRATCH), 'TMP': str(SCRATCH),
            'TEMP': str(SCRATCH)}
    exe = executables[0]['executable']
    prec = run('%s-panic-run' % arm, [exe], tenv, 'other', cwd=SRC, timeout=60, expect_zero=False)
    text = (ROOT / prec['stdout']).read_text(errors='replace') + '\n' + \
        (ROOT / prec['stderr']).read_text(errors='replace')
    bt = analyse_backtrace(text)
    bt['exit'] = prec['exit']; bt['expected_exit'] = 101
    bt['executable'] = exe.replace(str(ROOT), '<root>'); bt['executable_sha256'] = sha256(exe)
    bt['dsym_next_to_executable'] = os.path.isdir(exe + '.dSYM')
    arm_rec['backtrace'] = bt; save()
    boundary(arm + ' end')
    owned_check(arm + ' final')

def unit_map(name):
    rec = next((c for c in result['calls'] if c['name'] == name), None)
    if rec is None or 'rustc_units' not in rec:
        return None
    return {u['key']: u for u in rec['rustc_units']}

def check_arm_control(arm, rec):
    """Stop if the arm changes anything except its specified debug setting."""
    if arm == cfg['arm_order'][0]:
        return
    ref = unit_map(cfg['arm_order'][0] + '-cold-build'); units = unit_map(rec['name'])
    if ref is None or units is None or set(ref) != set(units):
        raise Stop('%s cold build unit set differs from the reference arm' % arm)
    for k, u in units.items():
        r = ref[k]
        if u['settings_except_debug_sha256'] != r['settings_except_debug_sha256']:
            raise Stop('%s unit %s changed a non-debug setting' % (arm, k))
        if arm == 'N' and not u['workspace'] and u['debug_flags'] != r['debug_flags']:
            raise Stop('unsupported: N package override changed dependency unit %s debug flags' % k)
        if arm == 'N' and u['workspace'] and u['debug_flags'].get('debuginfo') not in (None, '0'):
            raise Stop('unsupported: N package override left workspace unit %s debuginfo %s'
                       % (k, u['debug_flags'].get('debuginfo')))
        if arm == 'P' and {x: y for x, y in u['debug_flags'].items() if x != 'split-debuginfo'} != \
                {x: y for x, y in r['debug_flags'].items() if x != 'split-debuginfo'}:
            raise Stop('P unit %s changed a debug flag other than split-debuginfo' % k)

def compare_settings():
    """Record resolved debug flags per unit class for every arm's cold build."""
    out = {}
    for arm in cfg['arm_order']:
        units = unit_map(arm + '-cold-build')
        if units is None:
            continue
        classes = {}
        for u in units.values():
            cls = 'workspace' if u['workspace'] else ('dependency build script' if u['build_script'] else 'dependency')
            flags = json.dumps(u['debug_flags'], sort_keys=True)
            classes.setdefault(cls, {}).setdefault(flags, []).append(u['key'].split('|')[0] + ':' + u['key'].split('|')[1])
        rec = next(c for c in result['calls'] if c['name'] == arm + '-cold-build')
        out[arm] = {'unit_count': len(units),
                    'debug_flags_by_class': {c: {k: sorted(v) for k, v in m.items()} for c, m in classes.items()},
                    'workspace_member_artifacts': sorted({a['package_id'] for a in rec['artifact_profiles']
                                                          if a['workspace_member']})}
    result['settings_comparison'] = out; save()

ATTEMPT_1 = ROOT / 'result-attempt-1.json'

def resume_from_attempt_1():
    """Attempt 2 continues attempt 1's session: same packet, clock, counters, root.

    Attempt 1 stopped on a harness defect (CompletedProcess has no pid) after
    both version calls and U's cold build. Its calls and counts are carried
    forward unchanged; no setup or version call is repeated.
    """
    prior = json.loads(ATTEMPT_1.read_text())
    assert prior['packet'] == cfg and prior['outcome'] == 'partial'
    for k in ('calls', 'arms', 'storage', 'boundaries', 'notes', 'toolchain_identity', 'source'):
        result[k] = prior[k]
    counts.update(prior['counts']); result['counts'] = counts
    scratch_reference['free'] = prior['storage']['scratch_reference_free']
    result['attempts'] = [{'attempt': 1, 'session': prior['session'], 'outcome': prior['outcome'],
                           'stop_reason': prior['stop_reason'], 'counts': prior['counts'],
                           'raw_log_bytes': prior.get('raw_log_bytes')}]
    result['session'] = dict(prior['session'])
    result['session']['attempt_2_admitted_offset_s'] = offset()
    result['session']['attempt_2_admitted_utc'] = time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime())
    boundary('attempt 2 admitted')

def main():
    if ATTEMPT_1.exists():
        try:
            resume_from_attempt_1()
            measure_arm('U', resume=True)
            for arm in cfg['arm_order'][1:]:
                measure_arm(arm)
            result['outcome'] = 'complete'
        except Stop as reason:
            result['stop_reason'] = str(reason)
            if result['outcome'] == 'running':
                result['outcome'] = 'partial'
        except Exception as error:  # preserve partial result on harness failure
            result['stop_reason'] = 'harness error: %r' % (error,)
            result['outcome'] = 'partial'
        return finish()
    result['session'] = {'start_ns': START, 'deadline_ns': DEADLINE, 'clock': cfg['clock'],
                         'admitted_offset_s': offset(), 'admitted_utc': time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime())}
    SCRATCH.mkdir(mode=0o700, exist_ok=True)
    assert os.path.realpath(SCRATCH) == str(SCRATCH), 'scratch root is not canonical'
    boundary('admitted')
    try:
        b = cfg['toolchain']['bin']
        venv = {'PATH': b + ':/usr/bin:/bin', 'HOME': '/Users/tom', 'CARGO_HOME': cfg['host_cargo_home']}
        run('cargo-version', [b + '/cargo', '-vV'], venv, 'setup', timeout=60)
        run('rustc-version', [b + '/rustc', '-vV'], venv, 'setup', timeout=60)
        libdir = pathlib.Path(b).parent / 'lib'
        result['toolchain_identity'] = {
            'cargo_sha256': sha256(b + '/cargo'), 'rustc_sha256': sha256(b + '/rustc'),
            'rustc_driver': {d.name: sha256(d) for d in sorted(libdir.glob('librustc_driver-*.dylib'))}}
        save()
        run('clone', ['/usr/bin/git', 'clone', '--quiet', '--no-hardlinks', '--no-checkout',
                      cfg['source_repository'], SRC], {'PATH': '/usr/bin:/bin', 'HOME': '/Users/tom'}, 'other',
            timeout=120)
        git('checkout', '--quiet', '--detach', PIN, name='checkout')
        head = git('rev-parse', 'HEAD', 'HEAD^{tree}', name='rev-parse')
        manifest = (SRC / 'Cargo.toml').read_text()
        result['source'] = {'head_and_tree': (ROOT / head['stdout']).read_text().split(),
                            'lock_sha256': sha256(SRC / 'Cargo.lock'),
                            'manifest_sha256': sha256(SRC / 'Cargo.toml'),
                            'manifest_has_workspace_table': bool(re.search(r'^\[workspace', manifest, re.M)),
                            'manifest_package_name': re.search(r'^name = "([^"]+)"', manifest, re.M).group(1)}
        save()
        assert result['source']['head_and_tree'][0] == PIN, 'clone HEAD is not the pin'
        if result['source']['manifest_has_workspace_table'] or \
                result['source']['manifest_package_name'] != WORKSPACE_PACKAGE:
            raise Stop('unsupported: workspace membership is not the single package %s' % WORKSPACE_PACKAGE)
        scratch_reference['free'] = free_bytes(); result['storage']['scratch_reference_free'] = scratch_reference['free']
        owned_check('after setup')
        for arm in cfg['arm_order']:
            measure_arm(arm)
        result['outcome'] = 'complete'
    except Stop as reason:
        result['stop_reason'] = str(reason)
        if result['outcome'] == 'running':
            result['outcome'] = 'partial'
    except Exception as error:  # preserve partial result on harness failure
        result['stop_reason'] = 'harness error: %r' % (error,)
        result['outcome'] = 'partial'
    return finish()

def finish():
    try:
        compare_settings()
    except Exception as error:
        result['notes'].append('settings comparison failed: %r' % (error,))
    boundary('end')
    try:
        owned_check('end')
    except Stop as reason:
        result['notes'].append(str(reason))
    result['session']['end_offset_s'] = offset()
    result['raw_log_bytes'] = raw_bytes()
    save()
    print('OUTCOME', result['outcome'], result['stop_reason'], 'counts', counts, flush=True)
    return 0 if result['outcome'] == 'complete' else 2

if __name__ == '__main__':
    raise SystemExit(main())
