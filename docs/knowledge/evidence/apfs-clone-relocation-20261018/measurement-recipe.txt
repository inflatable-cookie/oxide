class Scale(ctypes.Structure):
    _fields_ = [("numer", ctypes.c_uint32), ("denom", ctypes.c_uint32)]


scale = Scale()
assert mach.mach_timebase_info(ctypes.byref(scale)) == 0


def now():

class Stop(Exception):
    """A bound or uncertain writer closure: stop effects and preserve resources."""


def rel_s(t=None):
    return ((now() if t is None else t) - start_ns) / 1e9


def save():
    (root / "result.json").write_text(json.dumps(result, indent=2) + "\n")


def host_free():
    v = os.statvfs("/private/tmp")
    return v.f_bavail * v.f_frsize


def log_bytes():
    return sum(p.stat().st_size for p in logs.glob("*") if p.is_file())


def owned():
    seen = set()
    total = 0
    files = 0
    for base, dirs, names in os.walk(root):
        for name in dirs + names:
            try:
                st = os.lstat(os.path.join(base, name))
            except FileNotFoundError:
                continue
            key = (st.st_dev, st.st_ino)
            if key in seen:
                continue
            seen.add(key)
            total += st.st_blocks * 512
            files += 1
    return total, files


def bounds(walk=True):
    if now() >= deadline_ns - 10_000_000_000:
        raise Stop("original 30-minute session deadline reached")
    if host_free() < HOST_RESERVE_BYTES + 512 * MIB:
        raise Stop("20GiB raw host reserve threatened")
    if log_bytes() > RAW_LOG_STOP_BYTES:
        raise Stop("32MiB raw log cap threatened")
    if walk:
        total, _ = owned()
        if total > SCRATCH_STOP_BYTES:
            raise Stop("4GiB owned scratch cap threatened: %d" % total)


def phase(name, **extra):
    total, files = owned()
    record = {
        "name": name,
        "rel_s": rel_s(),
        "utc": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
        "host_free_bytes": host_free(),
        "owned_bytes": total,
        "owned_entries": files,
        "loadavg": list(os.getloadavg()),
    }
    record.update(extra)
    result["phases"].append(record)
    save()
    print("PHASE", name, round(record["rel_s"], 1), flush=True)
    return record


def group_closure(pgid):
    try:
        os.killpg(pgid, 0)
    except OSError as error:
        if error.errno == errno.ESRCH:
            return "esrch-empty"
        return "uncertain-errno-%d" % error.errno
    return "uncertain-members-remain"


def run(name, argv, cwd, env, kind, env_record=None):
    bounds()
    if kind in counts:
        cap = EXPERIMENT_CALL_CAP if kind == "experiment" else SETUP_CALL_CAP
        if counts[kind] + 1 > cap:
            raise Stop("%s Cargo call cap reached before %s" % (kind, name))
        counts[kind] += 1
    out_log = logs / (name + ".stdout.log")
    err_log = logs / (name + ".stderr.log")
    started = now()
    print(name, "START", flush=True)
    with out_log.open("wb") as out, err_log.open("wb") as err:
        child = subprocess.Popen(argv, cwd=cwd, env=env, stdout=out, stderr=err, start_new_session=True)
        last_walk = 0
        try:
            while child.poll() is None:
                walk = now() - last_walk > 2_000_000_000
                if walk:
                    last_walk = now()
                bounds(walk=walk)
                time.sleep(0.2)
        except BaseException as error:
            os.killpg(child.pid, signal.SIGTERM)
            try:
                child.wait(timeout=10)
            except subprocess.TimeoutExpired:
                os.killpg(child.pid, signal.SIGKILL)
                child.wait()
            result["calls"].append({
                "name": name, "argv": argv, "kind": kind, "stopped": str(error),
                "closure": group_closure(child.pid), "start_rel_s": rel_s(started), "end_rel_s": rel_s(),
            })
            save()
            raise Stop("stopped owned group of %s: %s" % (name, error))
    closure = group_closure(child.pid)
    record = {
        "name": name,
        "argv": argv,
        "cwd": str(cwd),
        "kind": kind,
        "experiment_calls_after": counts["experiment"],
        "setup_calls_after": counts["setup"],
        "env_overrides": env_record or {},
        "start_rel_s": rel_s(started),
        "end_rel_s": rel_s(),
        "wall_seconds": (now() - started) / 1e9,
        "exit": child.returncode,
        "closure": closure,
        "stdout_log": out_log.name,
        "stdout_bytes": out_log.stat().st_size,
        "stderr_log": err_log.name,
        "stderr_bytes": err_log.stat().st_size,
    }
    result["calls"].append(record)
    save()
    print(name, "EXIT", child.returncode, round(record["wall_seconds"], 3), closure, flush=True)
    if closure != "esrch-empty":
        raise Stop("uncertain writer closure after %s: %s" % (name, closure))
    return record, out_log.read_text(errors="replace"), err_log.read_text(errors="replace")

def write_fixture(src):
    digests = {}
    for rel, text in FIXTURE.items():
        path = src / rel
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_text(text)
        digests[rel] = hashlib.sha256(text.encode()).hexdigest()
    return digests


def edit(path, old, new):
    before = path.read_bytes()
    assert before.count(old.encode()) == 1, "edit anchor not unique in %s" % path
    after = before.replace(old.encode(), new.encode())
    path.write_bytes(after)
    return {
        "path": str(path),
        "replace": [old, new],
        "before_sha256": hashlib.sha256(before).hexdigest(),
        "after_sha256": hashlib.sha256(after).hexdigest(),
        "utc": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
    }

STDERR_KEYS = ("Fresh", "Compiling", "Dirty", "Finished", "Executable", "error", "warning")


def parse_build(stdout, stderr):
    artifacts = []
    scripts = []
    for line in stdout.splitlines():
        if not line.startswith("{"):
            continue
        try:
            message = json.loads(line)
        except ValueError:
            continue
        reason = message.get("reason")
        package = message.get("package_id", "").split("#")[-1]
        if reason == "compiler-artifact":
            target = message.get("target", {})
            artifacts.append({
                "package": package,
                "target": target.get("name"),
                "kind": target.get("kind"),
                "test": message.get("profile", {}).get("test"),
                "fresh": message.get("fresh"),
                "executable": message.get("executable"),
                "filenames": message.get("filenames", []),
            })
        elif reason == "build-script-executed":
            scripts.append({
                "package": package,
                "out_dir": message.get("out_dir"),
                "linked_paths": message.get("linked_paths"),
                "linked_libs": message.get("linked_libs"),
            })
    status = []
    script_runs = []
    for line in stderr.splitlines():
        stripped = line.strip()
        if stripped.split(" ", 1)[0] in STDERR_KEYS:
            status.append(stripped[:400])
        if stripped.startswith("Running `") and "build-script-build" in stripped and "rustc" not in stripped.split("`")[1].split(" ")[0]:
            script_runs.append(stripped[:400])
    fresh_flags = [a["fresh"] for a in artifacts]
    return {
        "artifacts": artifacts,
        "build_scripts": scripts,
        "stderr_status_lines": status[:300],
        "build_script_runs": script_runs[:50],
        "artifact_count": len(artifacts),
        "fresh_count": sum(1 for f in fresh_flags if f),
        "not_fresh": ["%s/%s%s" % (a["package"], a["target"], " (test)" if a["test"] else "") for a in artifacts if not a["fresh"]],
        "all_fresh": bool(artifacts) and all(fresh_flags) and not script_runs,
    }


# ---------------------------------------------------------------- clone

def inventory(base):
    entries = {}
    for current, dirs, names in os.walk(base):
        dirs.sort()
        for name in sorted(dirs + names):
            path = os.path.join(current, name)
            st = os.lstat(path)
            kind = "dir" if os.path.isdir(path) and not os.path.islink(path) else ("link" if os.path.islink(path) else "file")
            entries[os.path.relpath(path, base)] = (kind, st.st_ino, st.st_nlink, st.st_size, st.st_mtime_ns, st.st_mode, st.st_blocks)
    return entries


def phys(path):
    try:
        fd = os.open(path, os.O_RDONLY)
    except OSError as error:
        return {"error": "open errno %d" % error.errno}
    try:
        out = fcntl.fcntl(fd, F_LOG2PHYS_EXT, struct.pack("=Iqq", 0, 1 << 20, 0))
        _, contig, devoffset = struct.unpack("=Iqq", out)
        return {"devoffset": devoffset, "contig": contig}
    except OSError as error:
        return {"error": "F_LOG2PHYS_EXT errno %d" % error.errno}
    finally:
        os.close(fd)


def sha(path):
    digest = hashlib.sha256()
    with open(path, "rb") as handle:
        for chunk in iter(lambda: handle.read(1 << 20), b""):
            digest.update(chunk)
    return digest.hexdigest()


def clone_tree(label, src, dst):
    """clonefile(2) one directory hierarchy; stop if the native clone fails."""
    bounds()
    before_inventory = inventory(src)
    logical = sum(e[6] * 512 for e in before_inventory.values())
    total, _ = owned()
    if total + logical + 512 * MIB > SCRATCH_CAP_BYTES:
        raise Stop("clone of %s would breach the 4GiB owned scratch cap (owned %d + clone %d)" % (label, total, logical))
    free_before = host_free()
    t0 = now()
    rc = libc.clonefile(str(src).encode(), str(dst).encode(), CLONE_NOFOLLOW)
    err = ctypes.get_errno()
    t1 = now()
    free_after = host_free()
    time.sleep(3)
    free_after_settle = host_free()
    record = {
        "operation": "clonefile(src, dst, CLONE_NOFOLLOW) on the directory root",
        "src": str(src),
        "dst": str(dst),
        "rc": rc,
        "errno": err if rc != 0 else 0,
        "wall_seconds": (t1 - t0) / 1e9,
        "entries": len(before_inventory),
        "files": sum(1 for e in before_inventory.values() if e[0] == "file"),
        "dirs": sum(1 for e in before_inventory.values() if e[0] == "dir"),
        "src_logical_allocated_bytes": logical,
        "host_free_before": free_before,
        "host_free_after": free_after,
        "host_free_after_3s": free_after_settle,
        "host_free_delta_bytes": free_before - free_after,
        "host_free_delta_3s_bytes": free_before - free_after_settle,
    }
    result["arms"].setdefault(label, {})["clone"] = record
    save()
    if rc != 0:
        raise Stop("clonefile failed for %s: errno %d (%s)" % (label, err, os.strerror(err)))
    after = inventory(dst)
    files = [rel for rel, e in before_inventory.items() if e[0] == "file"]
    same_set = set(after) == set(before_inventory)
    distinct_ino = sum(1 for rel in files if rel in after and after[rel][1] != before_inventory[rel][1])
    same_size = sum(1 for rel in files if rel in after and after[rel][3] == before_inventory[rel][3])
    same_mtime = sum(1 for rel in files if rel in after and after[rel][4] == before_inventory[rel][4])
    same_mode = sum(1 for rel in files if rel in after and after[rel][5] == before_inventory[rel][5])
    dir_same_mtime = sum(1 for rel, e in before_inventory.items() if e[0] == "dir" and rel in after and after[rel][4] == e[4])
    src_nlink = {}
    dst_nlink = {}
    for rel in files:
        src_nlink[before_inventory[rel][2]] = src_nlink.get(before_inventory[rel][2], 0) + 1
        if rel in after:
            dst_nlink[after[rel][2]] = dst_nlink.get(after[rel][2], 0) + 1
    sample = sorted((rel for rel in files if before_inventory[rel][3] > 0), key=lambda r: -before_inventory[r][3])[:PHYS_SAMPLE]
    phys_equal = 0
    phys_differ = 0
    phys_error = 0
    examples = []
    for rel in sample:
        a = phys(str(src / rel))
        b = phys(str(dst / rel))
        if "error" in a or "error" in b:
            phys_error += 1
        elif a["devoffset"] == b["devoffset"]:
            phys_equal += 1
        else:
            phys_differ += 1
        if len(examples) < 5:
            examples.append({"rel": rel, "size": before_inventory[rel][3], "src": a, "dst": b})
    hash_checked = 0
    hash_equal = 0
    for rel in sample[:40]:
        hash_checked += 1
        if sha(src / rel) == sha(dst / rel):
            hash_equal += 1
    control = None
    small = [rel for rel in sample if before_inventory[rel][3] <= MIB]
    if small:
        control_path = root / ("control-copy-" + label)
        with open(src / small[0], "rb") as handle, open(control_path, "wb") as out:
            out.write(handle.read())
            out.flush()
            os.fsync(out.fileno())
        control = {
            "rel": small[0],
            "operation": "plain read/write copy, fsync",
            "src": phys(str(src / small[0])),
            "copy": phys(str(control_path)),
        }
        control["copy_devoffset_differs"] = control["src"].get("devoffset") != control["copy"].get("devoffset")
        control_path.unlink()
    record.update({
        "path_set_identical": same_set,
        "dst_entries": len(after),
        "files_distinct_inode": distinct_ino,
        "files_same_size": same_size,
        "files_same_mtime_ns": same_mtime,
        "files_same_mode": same_mode,
        "dirs_same_mtime_ns": dir_same_mtime,
        "src_file_nlink_histogram": {str(k): v for k, v in sorted(src_nlink.items())},
        "dst_file_nlink_histogram": {str(k): v for k, v in sorted(dst_nlink.items())},
        "phys_sample_size": len(sample),
        "phys_sample_rule": "largest non-empty files, F_LOG2PHYS_EXT device offset of file offset 0",
        "phys_first_extent_equal": phys_equal,
        "phys_first_extent_differ": phys_differ,
        "phys_errors": phys_error,
        "phys_examples": examples,
        "sha256_checked": hash_checked,
        "sha256_equal": hash_equal,
        "plain_copy_control": control,
    })
    save()
    return record

def classify(rel, head):
    parts = rel.split(os.sep)
    name = parts[-1]
    if ".fingerprint" in parts:
        return "cargo-fingerprint"
    if name in ("output", "root-output", "stderr", "invoked.timestamp"):
        return "build-script-record"
    if "out" in parts[:-1]:
        return "build-script-OUT_DIR-content"
    if "incremental" in parts:
        return "rustc-incremental"
    if name.endswith(".d"):
        return "rustc-or-cargo-dep-info-d"
    if name.endswith(".rlib") or name.endswith(".rmeta"):
        return "rust-library"
    if name.endswith(".o"):
        return "object"
    if head[:4] in (b"\xcf\xfa\xed\xfe", b"\xca\xfe\xba\xbe"):
        return "mach-o-binary"
    return "other"


def search(label, bases, needle):
    bounds()
    started = now()
    target = needle.encode()
    overlap = len(target) - 1
    read = 0
    scanned = 0
    hits = []
    by_category = {}
    uncovered = []
    uncovered_count = 0
    for base in bases:
        for current, dirs, names in os.walk(base):
            dirs.sort()
            for name in sorted(names):
                path = os.path.join(current, name)
                if os.path.islink(path) or not os.path.isfile(path):
                    continue
                size = os.lstat(path).st_size
                rel = os.path.join(os.path.basename(base), os.path.relpath(path, base))
                if read + size > SEARCH_READ_CAP or now() >= deadline_ns - 240_000_000_000:
                    uncovered_count += 1
                    if len(uncovered) < 50:
                        uncovered.append(rel)
                    continue
                count = 0
                first = None
                head = b""
                tail = b""
                offset = 0
                with open(path, "rb") as handle:
                    while True:
                        chunk = handle.read(8 << 20)
                        if not chunk:
                            break
                        if offset == 0:
                            head = chunk[:4]
                        window = tail + chunk
                        position = window.find(target)
                        while position != -1:
                            count += 1
                            if first is None:
                                absolute = offset - len(tail) + position
                                first = {"offset": absolute, "context": window[max(0, position - 48):position + len(target) + 64].decode("utf-8", "replace")}
                            position = window.find(target, position + 1)
                        tail = window[-overlap:] if overlap else b""
                        offset += len(chunk)
                read += size
                scanned += 1
                if count:
                    category = classify(os.path.relpath(path, base), head)
                    entry = by_category.setdefault(category, {"files": 0, "occurrences": 0})
                    entry["files"] += 1
                    entry["occurrences"] += count
                    if len(hits) < SEARCH_HIT_LIST_CAP:
                        hits.append({"rel": rel, "category": category, "size": size, "occurrences": count, "first": first})
    record = {
        "needle": needle,
        "bases": [str(b) for b in bases],
        "scanned_files": scanned,
        "scanned_bytes": read,
        "uncovered_files": uncovered_count,
        "uncovered_examples": uncovered,
        "hit_files": sum(v["files"] for v in by_category.values()),
        "hit_list_truncated": sum(v["files"] for v in by_category.values()) > len(hits),
        "by_category": by_category,
        "hits": hits,
        "wall_seconds": (now() - started) / 1e9,
    }
    (logs / ("search-" + label + ".json")).write_text(json.dumps(record, indent=1) + "\n")
    summary = {k: v for k, v in record.items() if k != "hits"}
    summary["detail_log"] = "search-" + label + ".json"
    return summary


# ---------------------------------------------------------------- executables

def run_exes(label, arm_dir, src, build, expected):
    outcomes = []
    for artifact in build["artifacts"]:
        exe = artifact.get("executable")
        if not exe:
            continue
        is_bin = artifact["kind"] == ["bin"] and not artifact["test"]
        is_it = artifact["kind"] == ["test"]
        if not (is_bin or is_it):
            continue
        env = {"PATH": BASE_PATH, "PROBE_EXPECT": expected, "TMPDIR": str(tmp), "TMP": str(tmp), "TEMP": str(tmp), "OXIDE_TASK_SCRATCH_ROOT": str(tmp)}
        name = "%s-exe-%s" % (label, "bin" if is_bin else "it")
        record, stdout, _ = run(name, [exe], src, env, "exe", {"PROBE_EXPECT": expected})
        if is_bin:
            ok = record["exit"] == 0 and stdout.strip() == expected
        else:
            ok = record["exit"] == 0 and "test result: ok. 1 passed" in stdout
        outcomes.append({"name": name, "executable": exe, "exit": record["exit"], "stdout_first_line": stdout.strip().splitlines()[:1], "pass": ok})
    return {"expected": expected, "runs": outcomes, "all_pass": bool(outcomes) and all(o["pass"] for o in outcomes) and len(outcomes) == 2}

def copy_registry(lock_text):
    cargo_source = pathlib.Path(cfg["host_cargo_home"])
    cargo_home = root / "cargo-home"
    packages = []
    current = {}
    for line in lock_text.splitlines():
        if line.startswith("[[package]]"):
            if current:
                packages.append(current)
            current = {}
        for key in ("name", "version", "source"):
            match = re.match(r'^%s = "([^"]+)"' % key, line)
            if match:
                current[key] = match.group(1)
    if current:
        packages.append(current)
    cache_root = cargo_source / "registry" / "cache"
    src_root = cargo_source / "registry" / "src"
    index_root = cargo_source / "registry" / "index"
    items = [(c, "registry/index/%s/config.json" % c.parent.name) for c in sorted(index_root.glob("*/config.json"))]
    for package in packages:
        if "registry" not in package.get("source", ""):
            continue
        name, version = package["name"], package["version"]
        for crate in sorted(cache_root.glob("*/%s-%s.crate" % (name, version)))[:1]:
            items.append((crate, "registry/cache/%s/%s" % (crate.parent.name, crate.name)))
        for source_dir in sorted(src_root.glob("*/%s-%s" % (name, version)))[:1]:
            items.append((source_dir, "registry/src/%s/%s" % (source_dir.parent.name, source_dir.name)))
        for cached in sorted(index_root.glob("*/.cache/**/%s" % name)):
            items.append((cached, "registry/index/%s" % cached.relative_to(index_root)))
    copied = 0
    crates = {}
    for source, rel in items:
        assert not source.is_symlink(), "registry input is a symlink"
        pairs = []
        if source.is_file():
            pairs.append((source, cargo_home / rel))
        else:
            for current_dir, _, names in os.walk(source):
                for name in names:
                    item = pathlib.Path(current_dir) / name
                    assert not item.is_symlink(), "registry input tree contains a symlink"
                    pairs.append((item, cargo_home / rel / item.relative_to(source)))
        for item, destination in pairs:
            copied += item.stat().st_size
            assert copied <= REGISTRY_COPY_CAP, "registry copy cap reached"
            destination.parent.mkdir(parents=True, exist_ok=True)
            with open(item, "rb") as handle, open(destination, "wb") as out:
                out.write(handle.read())
        if source.suffix == ".crate":
            crates[rel] = sha(source)
    return {"operation": "read-only byte copy of locked registry inputs into the task CARGO_HOME", "copied_bytes": copied, "items": len(items), "crate_sha256": dict(sorted(crates.items()))}


def restore(label):
    paths = result["arms"].get(label, {}).get("paths")
    if not paths:
        return
    moved = pathlib.Path(paths["build_a_moved"])
    original = pathlib.Path(paths["build_a"])
    if moved.exists() and not original.exists():
        os.rename(moved, original)
        result["arms"][label]["restored_a"] = {"from": str(moved), "to": str(original), "a_exists": original.exists(), "rel_s": rel_s()}
        save()
