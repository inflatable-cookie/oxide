#!/usr/bin/env python3
"""Disposable submission recipe for oxide#081 (APFS clone relocation probe).

Starts the one original session clock, creates one fresh task-owned root under
/private/tmp, binds the pinned source and the exact measurement recipe into
packet.json, then submits the single heavy-admitted run. Not a product file and
not a reusable launcher.
"""
import ctypes
import hashlib
import json
import os
import pathlib
import shlex
import shutil
import subprocess
import sys
import tempfile
import time
import uuid

lib = ctypes.CDLL("/usr/lib/libSystem.B.dylib")
lib.mach_continuous_time.restype = ctypes.c_uint64


class Scale(ctypes.Structure):
    _fields_ = [("numer", ctypes.c_uint32), ("denom", ctypes.c_uint32)]


scale = Scale()
assert lib.mach_timebase_info(ctypes.byref(scale)) == 0
START_NS = lib.mach_continuous_time() * scale.numer // scale.denom
UTC_START = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())

STAGE = pathlib.Path(__file__).resolve().parent
REPO = pathlib.Path.cwd()
SOURCE_HEAD = "dc84d039cd397c8292f37f4201e3916fa2ecd1d3"
SOURCE_TREE = "448bed9adba200dc4489060637a3bdb05addaddf"
CALLER = "oxide:worker:081-apfs-clone-relocation"
HOST_CARGO_HOME = "/Users/tom/.cargo"
STABLE_BIN = "/Users/tom/.rustup/toolchains/1.97.1-aarch64-apple-darwin/bin"
NIGHTLY_BIN = "/Users/tom/.rustup/toolchains/nightly-2026-08-05-aarch64-apple-darwin/bin"


def git(*args):
    return subprocess.run(
        ["/usr/bin/git", *args], cwd=REPO, capture_output=True, text=True, check=True
    ).stdout.strip()


assert git("rev-parse", SOURCE_HEAD + "^{tree}") == SOURCE_TREE, "pinned tree mismatch"

root = pathlib.Path(tempfile.mkdtemp(prefix="oxide-clone-probe.", dir="/private/tmp")).resolve()
assert str(root).startswith("/private/tmp/"), "root is not canonical /private/tmp"
os.chmod(root, 0o700)
(root / "raw").mkdir()
script = root / "measure.py"
shutil.copyfile(STAGE / "measurement-recipe.txt", script)
packet = {
    "root": str(root),
    "request_id": str(uuid.uuid4()),
    "caller": CALLER,
    "session_start_ns": START_NS,
    "session_seconds_cap": 1800,
    "utc_start": UTC_START,
    "clock": "raw mach_continuous_time converted by mach_timebase_info, same boot; starts before any staging",
    "source_repository": str(REPO),
    "source_head": SOURCE_HEAD,
    "source_tree": SOURCE_TREE,
    "recipe_sha256": hashlib.sha256(script.read_bytes()).hexdigest(),
    "host_cargo_home": HOST_CARGO_HOME,
    "stable_bin": STABLE_BIN,
    "nightly_bin": NIGHTLY_BIN,
    "submitted_utc": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
}
(root / "packet.json").write_text(json.dumps(packet, indent=2) + "\n")
(root / "effigy.toml").write_text(
    "[manifest]\nminimum_effigy_version = \"0.14.0\"\n"
    "[tasks.clone-probe]\nadmission = \"heavy\"\nrun = "
    + json.dumps("exec " + shlex.join([sys.executable, str(script)]))
    + "\n"
)
print("SUBMIT root", root, "request", packet["request_id"], flush=True)
os.chdir(root)
os.environ["EFFIGY_CALLER"] = CALLER
os.execvp("effigy", ["effigy", "--host-run-request-id", packet["request_id"], "clone-probe"])
