#!/usr/bin/env python3
"""Disposable measurement recipe, not an Oxide product or reusable launcher."""
import ctypes, hashlib, json, os, pathlib, re, shutil, signal, subprocess, sys, time

P = pathlib.Path(__file__).parent
cfg = json.loads((P / 'packet.json').read_text())
root = pathlib.Path(cfg['root']); mount = root / 'm'; logs = root / 'raw'
lib = ctypes.CDLL('/usr/lib/libSystem.B.dylib')
lib.mach_continuous_time.restype = ctypes.c_uint64
class Scale(ctypes.Structure):
    _fields_ = [('numer', ctypes.c_uint32), ('denom', ctypes.c_uint32)]
scale = Scale(); assert lib.mach_timebase_info(ctypes.byref(scale)) == 0
def now(): return lib.mach_continuous_time() * scale.numer // scale.denom
deadline = cfg['deadline_ns']; count = 0
result = {'packet': cfg, 'calls': [], 'measurements': [], 'outcome': 'running',
          'writes': 'unknown; occupancy and time -l block operations are not SSD writes'}
def save():
    (root / 'result.json').write_text(json.dumps(result, indent=2) + '\n')
def bounds():
    assert now() < deadline - 5_000_000_000, 'original 20-minute deadline reached'
    v = os.statvfs('/private/var/tmp')
    assert v.f_bavail * v.f_frsize >= 20 * 1024**3, '20GiB raw host reserve reached'
    if os.path.ismount(mount):
        v = os.statvfs(mount)
        assert (v.f_blocks - v.f_bfree) * v.f_frsize < 14 * 1024**3, '14GiB image threshold reached'
    assert sum(p.stat().st_size for p in logs.glob('*') if p.is_file()) <= 8 * 1024**2, '8MiB raw log cap reached'
def run(name, argv, cwd=None, env=None, counted=False, expected=0):
    global count
    bounds()
    if counted:
        count += 1; assert count <= 10
    log = logs / (name + '.log')
    t = now(); print(name, 'START', flush=True)
    with log.open('wb') as output:
        child = subprocess.Popen(argv, cwd=cwd, env=env, stdout=output, stderr=subprocess.STDOUT, start_new_session=True)
        try:
            while child.poll() is None:
                bounds(); time.sleep(0.25)
        except BaseException:
            os.killpg(child.pid, signal.SIGTERM)
            try: child.wait(timeout=2)
            except subprocess.TimeoutExpired: os.killpg(child.pid, signal.SIGKILL); child.wait()
            raise
    rec = {'name': name, 'argv': argv, 'counted': counted, 'wall_seconds': (now()-t)/1e9,
           'exit': child.returncode, 'log': str(log), 'utc_end': time.strftime('%Y-%m-%dT%H:%M:%SZ',time.gmtime())}
    result['calls'].append(rec); save(); print(name, 'EXIT', child.returncode, rec['wall_seconds'], flush=True)
    assert child.returncode == expected, name + ' unexpected exit; see retained raw log'
    return log.read_text(errors='replace')
def inventory(mode, step):
    target = mount / ('target-' + mode); seen = set(); allocated = 0; dirs = {}
    for directory in ('deps', 'incremental'):
        files = objects = apparent = 0
        for base, _, names in os.walk(target / 'debug' / directory):
            for name in names:
                
                if files % 1024 == 0: bounds()
                st = os.lstat(pathlib.Path(base) / name); files += 1
                assert files <= 65536, 'inventory file cap'
                objects += name.endswith('.o'); apparent += st.st_size
                key = (st.st_dev, st.st_ino)
                if key not in seen: seen.add(key); allocated += st.st_blocks * 512
        dirs[directory] = {'files': files, 'objects': objects, 'apparent_bytes': apparent}
    result['measurements'].append({'mode': mode, 'step': step, 'directories': dirs,
        'deps_incremental_unique_inode_allocated_bytes': allocated})
    run(mode+'-'+step+'-du', ['/usr/bin/du','-sk',str(target)])
    save()

try:
    assert os.environ.get('HOST_RUN_TOKEN') and os.environ.get('HOST_RUN_ID'), 'missing genuine admission'
    assert pathlib.Path.cwd().resolve() == root.resolve(), 'incorrect genuine root'
    result['host_run_id'] = os.environ['HOST_RUN_ID']
    result['inner_entry_ns'] = now(); assert now() >= cfg['start_ns']
    logs.mkdir(exist_ok=True); save(); bounds()
    run('image-create', ['/usr/bin/hdiutil','create','-size','15g','-type','SPARSE','-fs','APFS','-volname','OxideComparison',str(root/'comparison.sparseimage')])
    mount.mkdir()
    run('image-attach', ['/usr/bin/hdiutil','attach','-nobrowse','-mountpoint',str(mount),str(root/'comparison.sparseimage')])
    cargo_home = mount / 'cargo-home'; cargo_home.mkdir()
    copied = 0
    for item in cfg['registry_inputs']:
        bounds(); src = pathlib.Path(item['src']); assert not src.is_symlink()
        copied += src.stat().st_size; assert copied <= 512 * 1024**2
        dest = cargo_home / item['rel']; dest.parent.mkdir(parents=True,exist_ok=True); shutil.copyfile(src,dest)
    result['registry_copied_bytes'] = copied; save()
    source = mount / 'src'
    run('clone', ['/usr/bin/git','clone','--no-local','--no-checkout',cfg['source_repository'],str(source)])
    run('checkout', ['/usr/bin/git','checkout','--detach',cfg['source_head']],cwd=source)
    tree = run('tree',['/usr/bin/git','rev-parse','HEAD^{tree}'],cwd=source).strip(); assert tree == cfg['source_tree']
    assert hashlib.sha256((source/'Cargo.lock').read_bytes()).hexdigest() == cfg['lock_sha256']
    for base in (source,mount,root,pathlib.Path('/private/var/tmp'),pathlib.Path('/private/var'),pathlib.Path('/private'),pathlib.Path('/'),cargo_home):
        for name in ('.cargo/config','.cargo/config.toml'):
            assert not (base/name).exists(), 'unexpected loaded config: '+str(base/name)
    env = os.environ.copy()
    for key in list(env):
        if key.startswith('CARGO_') or key in ('RUSTFLAGS','CARGO_ENCODED_RUSTFLAGS','RUSTC','RUSTC_WRAPPER','RUSTC_WORKSPACE_WRAPPER'):
            env.pop(key,None)
    toolbin = pathlib.Path('/Users/tom/.rustup/toolchains/1.97.1-aarch64-apple-darwin/bin')
    assert (toolbin/'cargo').is_file() and (toolbin/'rustc').is_file(), 'pinned installed toolchain absent'
    env.update(PATH=str(toolbin)+':'+os.environ['PATH'], CARGO_HOME=str(cargo_home), RUSTUP_TOOLCHAIN='1.97.1', CARGO_NET_OFFLINE='true')
    cargo = run('cargo-version',[str(toolbin/'cargo'),'-vV'],env=env,counted=True)
    rustc = run('rustc-version',[str(toolbin/'rustc'),'-vV'],env=env,counted=True)
    assert 'cargo 1.97.1' in cargo and 'c980f4866' in cargo
    assert 'release: 1.97.1' in rustc and '8bab26f4f' in rustc
    files = ['crates/nucleus-core/src/lib.rs','crates/nucleus-core/tests/packages.rs']
    for mode in ('off','on'):
        run(mode+'-load',['/usr/bin/uptime'])
        tmp = mount/('tmp-'+mode); tmp.mkdir()
        target = mount/('target-'+mode)
        env.update(CARGO_INCREMENTAL='0' if mode=='off' else '1', CARGO_TARGET_DIR=str(target),
            CARGO_PROFILE_DEV_SPLIT_DEBUGINFO='unpacked', CARGO_PROFILE_TEST_SPLIT_DEBUGINFO='unpacked',
            TMPDIR=str(tmp), TMP=str(tmp), TEMP=str(tmp))
        for step in ('C0','E1','E2','U'):
            if step in ('E1','E2'):
                with (source/files[0 if step=='E1' else 1]).open('a') as f: f.write('\n// oxide direct comparison '+step+'\n')
            text = run(mode+'-'+step,['/usr/bin/time','-l','effigy','check:core:test','--','-p','nucleus-core','--test','packages','-v','--','--exact','committed_workflow_packages_parse'],cwd=source,env=env,counted=True)
            assert 'test committed_workflow_packages_parse ... ok' in text and 'test result: ok. 1 passed; 0 failed;' in text
            if step == 'U': assert not re.search(r'^\s*Compiling\s+nucleus-core\s',text,re.M), 'unchanged unit rebuilt'
            inventory(mode,step)
        candidates=[]
        for line in text.splitlines():
            if 'Running `' in line and '--exact committed_workflow_packages_parse' in line:
                candidate=line.split('Running `',1)[1].split(' --exact',1)[0]
                path=pathlib.Path(candidate)
                if not path.is_absolute(): path=source/path
                candidates.append(path)
        assert len(candidates)==1 and candidates[0].is_file(), 'exact reported test executable unavailable'
        (source/'packages').rename(source/'packages.oxide-moved')
        panic_env=env.copy();panic_env['RUST_BACKTRACE']='1'
        panic=run(mode+'-backtrace',[str(candidates[0]),'--exact','committed_workflow_packages_parse'],cwd=source,env=panic_env,expected=101)
        result.setdefault('backtrace',{})[mode]={'frame_fileline':bool(re.search(r'^\s+at\s+.*tests/packages\.rs:\d+',panic,re.M)), 'expected_panic':'packages directory' in panic}
        (source/'packages.oxide-moved').rename(source/'packages')
        run(mode+'-restore',['/usr/bin/git','checkout','--']+files,cwd=source)
        assert not run(mode+'-clean',['/usr/bin/git','status','--porcelain'],cwd=source).strip()
    result['outcome']='complete'
except BaseException as exc:
    result['outcome']='partial';result['error']=str(exc)
finally:
    result['count_delivered']=count;result['end_ns']=now();result['total_seconds']=(now()-cfg['start_ns'])/1e9
    save();print(json.dumps({'outcome':result['outcome'],'count':count,'result':str(root/'result.json'),'error':result.get('error')}),flush=True)
sys.exit(0 if result['outcome']=='complete' else 1)
