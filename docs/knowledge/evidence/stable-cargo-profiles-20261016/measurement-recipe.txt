#!/usr/bin/env python3
"""Disposable one-shot comparison recipe for oxide#068. Not a product file.

Runs inside one genuine Effigy heavy-admitted run. Creates exactly one fresh
16 GiB APFS payload volume under the task-owned root, copies the pushed
implementation source into it, and runs the finite baseline / oxide-light /
oxide-full-debug Cargo sequence with monotonic timing and allocation
inventory. It never touches the canonical checkout's target directory and
reports a bounded partial/no-go rather than resetting the clock or the caps.
"""
import ctypes
import hashlib
import json
import os
import pathlib
import re
import shlex
import shutil
import signal
import subprocess
import sys
import time

STAGE = pathlib.Path(__file__).resolve().parent
cfg = json.loads((STAGE / "packet.json").read_text())
root = pathlib.Path(cfg["root"])
mount = root / "m"
logs = root / "raw"

SESSION_SECONDS = 20 * 60
HOST_RESERVE_BYTES = 20 * 1024**3
IMAGE_STOP_BYTES = 14 * 1024**3
RAW_LOG_CAP_BYTES = 32 * 1024**2
OUTSIDE_EVIDENCE_CAP_BYTES = 64 * 1024**2
CARGO_CALL_CAP = 8
REGISTRY_COPY_CAP = 512 * 1024**2
IMAGE_SIZE = "16g"

EDIT_MARKER = (
    b"\n/// Comparison-only marker added by the identical oxide#068 edit in both source copies.\n"
    b"pub const PROFILE_COMPARISON_MARKER: u32 = 68;\n"
)

lib = ctypes.CDLL("/usr/lib/libSystem.B.dylib")
lib.mach_continuous_time.restype = ctypes.c_uint64


class Scale(ctypes.Structure):
    _fields_ = [("numer", ctypes.c_uint32), ("denom", ctypes.c_uint32)]


scale = Scale()
assert lib.mach_timebase_info(ctypes.byref(scale)) == 0


def now():
    return lib.mach_continuous_time() * scale.numer // scale.denom


start_ns = now()
deadline_ns = start_ns + SESSION_SECONDS * 1_000_000_000
count = 0
result = {
    "packet": cfg,
    "session_start_ns": start_ns,
    "deadline_ns": deadline_ns,
    "session_seconds_cap": SESSION_SECONDS,
    "clock": "raw mach_continuous_time converted by mach_timebase_info, same boot",
    "calls": [],
    "measurements": [],
    "outcome": "running",
    "writes": "unknown; occupancy and time -l block operations are not SSD writes",
}


def save():
    (root / "result.json").write_text(json.dumps(result, indent=2) + "\n")


def log_bytes():
    return sum(p.stat().st_size for p in logs.glob("*") if p.is_file())


def bounds():
    assert now() < deadline_ns - 5_000_000_000, "original 20-minute session deadline reached"
    v = os.statvfs("/private/tmp")
    assert v.f_bavail * v.f_frsize >= HOST_RESERVE_BYTES, "20GiB raw host reserve reached"
    if os.path.ismount(mount):
        v = os.statvfs(mount)
        assert (v.f_blocks - v.f_bfree) * v.f_frsize < IMAGE_STOP_BYTES, "14GiB image threshold reached"
    assert log_bytes() <= RAW_LOG_CAP_BYTES, "32MiB raw log cap reached"
    assert outside_evidence_bytes() <= OUTSIDE_EVIDENCE_CAP_BYTES, "64MiB outside-volume evidence cap reached"


def outside_evidence_bytes():
    total = 0
    for base, dirs, names in os.walk(root):
        if pathlib.Path(base) == mount:
            dirs[:] = []
            continue
        for name in names:
            path = pathlib.Path(base) / name
            if path.is_symlink() or path.name.endswith(".sparseimage"):
                continue
            st = path.stat()
            total += max(st.st_size, st.st_blocks * 512)
    return total


def run(name, argv, cwd=None, env=None, counted=False, expected=0, parse_time=False):
    global count
    bounds()
    if counted:
        count += 1
        assert count <= CARGO_CALL_CAP, "Cargo invocation cap reached"
    log = logs / (name + ".log")
    started = now()
    print(name, "START", flush=True)
    with log.open("wb") as output:
        child = subprocess.Popen(
            argv,
            cwd=cwd,
            env=env,
            stdout=output,
            stderr=subprocess.STDOUT,
            start_new_session=True,
        )
        try:
            while child.poll() is None:
                bounds()
                time.sleep(0.25)
        except BaseException:
            os.killpg(child.pid, signal.SIGTERM)
            try:
                child.wait(timeout=5)
            except subprocess.TimeoutExpired:
                os.killpg(child.pid, signal.SIGKILL)
                child.wait()
            raise
    text = log.read_text(errors="replace")
    record = {
        "name": name,
        "argv": argv,
        "counted_cargo_call": counted,
        "wall_seconds": (now() - started) / 1e9,
        "exit": child.returncode,
        "log": str(log),
        "log_bytes": log.stat().st_size,
        "utc_end": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
    }
    if parse_time:
        match = re.search(r"^\s*([0-9.]+) real", text, re.M)
        if match:
            record["time_l_real_seconds"] = float(match.group(1))
        match = re.search(r"^\s*(\d+)\s+maximum resident set size", text, re.M)
        if match:
            record["time_l_max_rss_bytes"] = int(match.group(1))
    result["calls"].append(record)
    save()
    print(name, "EXIT", child.returncode, round(record["wall_seconds"], 3), flush=True)
    assert child.returncode == expected, name + " unexpected exit; see retained raw log"
    return text


def rustc_units(text):
    units = []
    for line in text.splitlines():
        match = re.search(r"Running `([^`]*)`", line)
        if not match:
            continue
        argv = shlex.split(match.group(1))
        if not argv or os.path.basename(argv[0]) != "rustc":
            continue
        crate = None
        crate_types = []
        source = None
        flags = {}
        index = 1
        while index < len(argv):
            arg = argv[index]
            if arg == "--crate-name" and index + 1 < len(argv):
                crate = argv[index + 1]
                index += 2
                continue
            if arg == "--crate-type" and index + 1 < len(argv):
                crate_types.append(argv[index + 1])
                index += 2
                continue
            if arg == "-C" and index + 1 < len(argv):
                raw = argv[index + 1]
                index += 2
            elif arg.startswith("-C") and len(arg) > 2:
                raw = arg[2:]
                index += 1
            else:
                if not arg.startswith("-") and arg.endswith(".rs") and source is None:
                    source = arg
                index += 1
                continue
            key, _, value = raw.partition("=")
            flags[key] = value if value else True
        if crate is not None:
            units.append(
                {"crate": crate, "source": source, "crate_types": crate_types, "flags": flags}
            )
    return units


def package_events(text):
    compiled = re.findall(r"^\s*Compiling (\S+) v(\S+)", text, re.M)
    fresh = re.findall(r"^\s*Fresh (\S+) v(\S+)", text, re.M)
    return {
        "compiled": ["%s %s" % item for item in compiled],
        "fresh": ["%s %s" % item for item in fresh],
        "rustc_invocations": len(rustc_units(text)),
    }


def profile_settings(text):
    records = []
    for unit in rustc_units(text):
        flags = unit["flags"]
        records.append(
            {
                "crate": unit["crate"],
                "source": unit["source"],
                "crate_types": unit["crate_types"],
                "debuginfo": flags.get("debuginfo"),
                "strip": flags.get("strip"),
                "incremental": bool(flags.get("incremental")),
                "opt_level": flags.get("opt-level"),
                "panic": flags.get("panic"),
                "debug_assertions": flags.get("debug-assertions"),
                "overflow_checks": flags.get("overflow-checks"),
            }
        )
    records.sort(key=lambda record: (record["crate"], record["source"] or ""))
    return records


PROBE_OK = "test profile_probe_keeps_workspace_frames_and_unwind ... ok"


def require_probe(text, name):
    assert PROBE_OK in text, name + " did not run the profile probe"
    assert "test result: ok. 1 passed" in text, name + " did not pass the profile probe"


def inventory(mode, step, target, profile_dir):
    seen = set()
    allocated = 0
    directories = {}
    for directory in ("deps", "incremental"):
        base_dir = target / profile_dir / directory
        files = objects = apparent = 0
        if base_dir.is_dir():
            for base, _, names in os.walk(base_dir):
                for name in names:
                    if files % 1024 == 0:
                        bounds()
                    path = pathlib.Path(base) / name
                    st = path.lstat()
                    files += 1
                    assert files <= 65536, "inventory file cap"
                    objects += name.endswith(".o")
                    apparent += st.st_size
                    key = (st.st_dev, st.st_ino)
                    if key not in seen:
                        seen.add(key)
                        allocated += st.st_blocks * 512
        directories[directory] = {
            "files": files,
            "objects": objects,
            "apparent_bytes": apparent,
        }
    run(mode + "-" + step + "-du", ["/usr/bin/du", "-sk", str(target)])
    result["measurements"].append(
        {
            "mode": mode,
            "step": step,
            "profile_dir": profile_dir,
            "directories": directories,
            "deps_incremental_unique_inode_allocated_bytes": allocated,
        }
    )
    save()


def clone_source(destination):
    run("clone-%s" % destination.name, ["/usr/bin/git", "clone", "--no-local", "--no-checkout", cfg["source_repository"], str(destination)])
    run(
        "checkout-%s" % destination.name,
        ["/usr/bin/git", "checkout", "--detach", cfg["source_head"]],
        cwd=destination,
    )
    tree = run("tree-%s" % destination.name, ["/usr/bin/git", "rev-parse", "HEAD^{tree}"], cwd=destination).strip()
    assert tree == cfg["source_tree"], "cloned tree mismatch for " + str(destination)
    return destination


def apply_edit(source):
    path = source / "src" / "lib.rs"
    before = path.read_bytes()
    path.write_bytes(before + EDIT_MARKER)
    return {
        "before_sha256": hashlib.sha256(before).hexdigest(),
        "after_sha256": hashlib.sha256(path.read_bytes()).hexdigest(),
    }


def registry_inputs(lock_text, cargo_source):
    packages = []
    current = {}
    for line in lock_text.splitlines():
        if line.startswith("[[package]]"):
            if current:
                packages.append(current)
            current = {}
        name = re.match(r'^name = "([^"]+)"', line)
        version = re.match(r'^version = "([^"]+)"', line)
        source = re.match(r'^source = "([^"]+)"', line)
        if name:
            current["name"] = name.group(1)
        if version:
            current["version"] = version.group(1)
        if source:
            current["source"] = source.group(1)
    if current:
        packages.append(current)
    cache_root = cargo_source / "registry" / "cache"
    src_root = cargo_source / "registry" / "src"
    index_root = cargo_source / "registry" / "index"
    items = []
    for config in sorted(index_root.glob("*/config.json")):
        items.append((config, "registry/index/%s/config.json" % config.parent.name))
    for package in packages:
        name = package.get("name")
        version = package.get("version")
        if not name or not version or "registry" not in package.get("source", ""):
            continue
        crate = sorted(cache_root.glob("*/%s-%s.crate" % (name, version)))
        source_dir = sorted(src_root.glob("*/%s-%s" % (name, version)))
        index_cache = sorted(index_root.glob("*/.cache/**/%s" % name))
        if crate:
            items.append((crate[0], "registry/cache/%s/%s" % (crate[0].parent.name, crate[0].name)))
        if source_dir:
            items.append((source_dir[0], "registry/src/%s/%s" % (source_dir[0].parent.name, source_dir[0].name)))
        for cached in index_cache:
            items.append((cached, "registry/index/%s" % cached.relative_to(index_root)))
    return items


try:
    assert os.environ.get("HOST_RUN_TOKEN") and os.environ.get("HOST_RUN_ID"), "missing genuine Effigy admission"
    assert pathlib.Path.cwd().resolve() == root.resolve(), "incorrect genuine run root"
    result["host_run_id"] = os.environ["HOST_RUN_ID"]
    result["effigy_executable"] = shutil.which("effigy")
    logs.mkdir(exist_ok=True)
    save()
    bounds()

    run(
        "image-create",
        ["/usr/bin/hdiutil", "create", "-size", IMAGE_SIZE, "-type", "SPARSE", "-fs", "APFS", "-volname", "OxideProfiles", str(root / "profiles.sparseimage")],
    )
    mount.mkdir()
    run("image-attach", ["/usr/bin/hdiutil", "attach", "-nobrowse", "-mountpoint", str(mount), str(root / "profiles.sparseimage")])
    volume = os.statvfs(mount)
    result["volume"] = {
        "declared_image_size": IMAGE_SIZE,
        "mount_total_bytes": volume.f_blocks * volume.f_frsize,
        "mount_available_bytes_at_start": volume.f_bavail * volume.f_frsize,
    }
    save()

    baseline_src = clone_source(mount / "src-baseline")
    light_src = clone_source(mount / "src-light")
    for source in (baseline_src, light_src):
        lock = (source / "Cargo.lock").read_bytes()
        result.setdefault("source_lock_sha256", {})[source.name] = hashlib.sha256(lock).hexdigest()
    result["edit_marker_sha256"] = hashlib.sha256(EDIT_MARKER).hexdigest()
    result["edit_marker_bytes"] = len(EDIT_MARKER)
    save()

    cargo_home = mount / "cargo-home"
    cargo_home.mkdir()
    copied = 0
    manifest = []
    for src, rel in registry_inputs((baseline_src / "Cargo.lock").read_text(), pathlib.Path(cfg["host_cargo_home"])):
        bounds()
        assert not src.is_symlink(), "registry input is a symlink"
        if src.is_file():
            copied += src.stat().st_size
            if src.suffix == ".crate":
                manifest.append((rel, hashlib.sha256(src.read_bytes()).hexdigest()))
            destination = cargo_home / rel
            destination.parent.mkdir(parents=True, exist_ok=True)
            shutil.copyfile(src, destination)
        else:
            for base, _, names in os.walk(src):
                for name in names:
                    bounds()
                    item = pathlib.Path(base) / name
                    assert not item.is_symlink(), "registry input tree contains a symlink"
                    copied += item.stat().st_size
                    destination = cargo_home / rel / item.relative_to(src)
                    destination.parent.mkdir(parents=True, exist_ok=True)
                    shutil.copyfile(item, destination)
        assert copied <= REGISTRY_COPY_CAP, "registry copy cap reached"
    result["registry_copy"] = {
        "copied_bytes": copied,
        "crate_sha256": dict(sorted(manifest)),
    }
    save()

    for base in (baseline_src, light_src, cargo_home, mount, root):
        for name in (".cargo/config", ".cargo/config.toml", "config", "config.toml"):
            assert not (base / name).exists(), "unexpected loaded Cargo config: %s" % (base / name)
    rustup_toolchain = pathlib.Path(cfg["toolchain_bin"])
    assert (rustup_toolchain / "cargo").is_file() and (rustup_toolchain / "rustc").is_file(), "pinned toolchain absent"

    env = os.environ.copy()
    for key in list(env):
        if key.startswith("CARGO_") or key in ("RUSTFLAGS", "CARGO_ENCODED_RUSTFLAGS", "RUSTC", "RUSTC_WRAPPER", "RUSTC_WORKSPACE_WRAPPER"):
            env.pop(key, None)
    env.update(
        PATH=str(rustup_toolchain) + ":" + os.environ["PATH"],
        CARGO_HOME=str(cargo_home),
        RUSTUP_TOOLCHAIN="1.97.1",
        CARGO_NET_OFFLINE="true",
        CARGO_INCREMENTAL="1",
    )
    result["environment"] = {
        "RUSTUP_TOOLCHAIN": env["RUSTUP_TOOLCHAIN"],
        "CARGO_HOME": env["CARGO_HOME"],
        "CARGO_NET_OFFLINE": env["CARGO_NET_OFFLINE"],
        "CARGO_INCREMENTAL": env["CARGO_INCREMENTAL"],
        "PATH_prefix": str(rustup_toolchain),
    }
    save()

    cargo_version = run("cargo-version", [str(rustup_toolchain / "cargo"), "-vV"], env=env, counted=True)
    rustc_version = run("rustc-version", [str(rustup_toolchain / "rustc"), "-vV"], env=env)
    assert "cargo 1.97.1" in cargo_version, "unexpected Cargo version"
    assert "release: 1.97.1" in rustc_version, "unexpected rustc version"
    result["toolchain_versions"] = {
        "cargo": cargo_version.strip().splitlines(),
        "rustc": rustc_version.strip().splitlines(),
    }
    save()

    def mode_env(mode):
        local = dict(env)
        temporary = mount / ("tmp-" + mode)
        temporary.mkdir(exist_ok=True)
        local.update(TMPDIR=str(temporary), TMP=str(temporary), TEMP=str(temporary))
        return local

    def cargo_test(call_name, source, profile, mode, expected_profile_dir):
        target = mount / ("target-" + mode)
        local = mode_env(mode)
        argv = ["/usr/bin/time", "-l", str(rustup_toolchain / "cargo"), "test", "--locked"]
        if profile is not None:
            argv += ["--profile", profile]
        argv += ["--test", "profile_backtrace", "-v"]
        local["CARGO_TARGET_DIR"] = str(target)
        text = run(call_name, argv, cwd=source, env=local, counted=True, parse_time=True)
        require_probe(text, call_name)
        assert (target / expected_profile_dir).is_dir(), call_name + " did not build the expected profile directory"
        result.setdefault("calls_detail", {})[call_name] = {
            "packages": package_events(text),
            "profile_settings": profile_settings(text),
        }
        inventory(mode, call_name, target, expected_profile_dir)
        return text

    def require_edit_rebuild(call_name):
        detail = result["calls_detail"][call_name]
        rebuilt = [
            unit for unit in detail["packages"]["compiled"] if unit.startswith("oxide-storage ")
        ]
        assert rebuilt, call_name + " did not rebuild the workspace after the identical edit"

    def run_mode(mode, source, profile, profile_dir):
        cargo_test(mode + "-C0", source, profile, mode, profile_dir)
        edit = apply_edit(source)
        result.setdefault("edits", {})[mode] = edit
        save()
        cargo_test(mode + "-E1", source, profile, mode, profile_dir)
        require_edit_rebuild(mode + "-E1")
        cargo_test(mode + "-U", source, profile, mode, profile_dir)
        detail = result["calls_detail"][mode + "-U"]
        assert not [
            unit for unit in detail["packages"]["compiled"] if unit.startswith("oxide-storage ")
        ], mode + " unchanged run rebuilt the workspace"
        return edit

    baseline_edit = run_mode("baseline", baseline_src, None, "debug")
    light_edit = run_mode("light", light_src, "oxide-light", "oxide-light")
    assert baseline_edit["before_sha256"] == light_edit["before_sha256"], "source copies differed before the edit"
    assert baseline_edit["after_sha256"] == light_edit["after_sha256"], "identical edit produced different sources"
    save()

    cargo_test("full-debug-C0", light_src, "oxide-full-debug", "full-debug", "oxide-full-debug")

    image = root / "profiles.sparseimage"
    stat = image.stat()
    result["image"] = {
        "logical_bytes": stat.st_size,
        "allocated_bytes": stat.st_blocks * 512,
    }
    result["raw_log_bytes"] = log_bytes()
    result["outside_evidence_bytes"] = outside_evidence_bytes()
    detach = run("image-detach", ["/usr/bin/hdiutil", "detach", str(mount)])
    result["detach"] = detach.strip()
    result["outcome"] = "complete"
except BaseException as exc:
    result["outcome"] = "partial"
    result["error"] = "%s: %s" % (type(exc).__name__, exc)
finally:
    result["count_delivered"] = count
    result["end_ns"] = now()
    result["total_seconds"] = (now() - start_ns) / 1e9
    result["utc_end"] = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
    try:
        save()
    except BaseException as save_error:
        print("result save failed: %s" % save_error, flush=True)
    print(json.dumps({"outcome": result["outcome"], "cargo_calls": count, "total_seconds": result["total_seconds"], "error": result.get("error")}), flush=True)

sys.exit(0 if result["outcome"] == "complete" else 1)
