#!/usr/bin/env python3
"""Disposable submission recipe for the oxide#068 profile comparison.

Creates one fresh task-owned root under /private/tmp, binds the pushed
implementation head and the exact measurement recipe into packet.json, then
submits the single heavy-admitted comparison run. Not a product file and not a
reusable launcher.
"""
import hashlib
import json
import os
import pathlib
import shlex
import shutil
import subprocess
import sys
import tempfile
import time
import uuid

STAGE = pathlib.Path(__file__).resolve().parent
REPO = pathlib.Path.cwd()
SOURCE_HEAD = "c762a731276c6677993002064ef5287676344fde"
SOURCE_TREE = "c3c98d78a59cf1b8ed8973f67a66404bd0554dfd"
CALLER = "oxide:worker:068-stable-profiles"
TOOLCHAIN_BIN = "/Users/tom/.rustup/toolchains/1.97.1-aarch64-apple-darwin/bin"
HOST_CARGO_HOME = "/Users/tom/.cargo"


def git(*args):
    return subprocess.run(
        ["/usr/bin/git", *args], cwd=REPO, capture_output=True, text=True, check=True
    ).stdout.strip()


head = git("rev-parse", "HEAD")
tree = git("rev-parse", "HEAD^{tree}")
assert head == SOURCE_HEAD, "workspace HEAD %s is not the measured implementation head" % head
assert tree == SOURCE_TREE, "workspace tree %s is not the measured implementation tree" % tree
assert not git("status", "--porcelain"), "workspace is not clean"

root = pathlib.Path(tempfile.mkdtemp(prefix="oxide-stable-profiles.", dir="/private/tmp"))
os.chmod(root, 0o700)
(root / "raw").mkdir()
script = root / "measure.py"
shutil.copyfile(STAGE / "measure.py", script)
packet = {
    "root": str(root),
    "request_id": str(uuid.uuid4()),
    "caller": CALLER,
    "source_repository": str(REPO),
    "source_head": head,
    "source_tree": tree,
    "lock_sha256": hashlib.sha256((REPO / "Cargo.lock").read_bytes()).hexdigest(),
    "recipe_sha256": hashlib.sha256(script.read_bytes()).hexdigest(),
    "host_cargo_home": HOST_CARGO_HOME,
    "toolchain_bin": TOOLCHAIN_BIN,
    "cargo_call_cap": 8,
    "session_seconds_cap": 1200,
    "image_size": "16g",
    "submitted_utc": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
    "note": "measured tree is the implementation commit; later commits are docs/evidence only",
}
(root / "packet.json").write_text(json.dumps(packet, indent=2) + "\n")
(root / "effigy.toml").write_text(
    "[manifest]\nminimum_effigy_version = \"0.14.0\"\n"
    "[tasks.comparison]\nadmission = \"heavy\"\nrun = "
    + json.dumps("exec " + shlex.join([sys.executable, str(script)]))
    + "\n"
)
print("SUBMIT root", root, "request", packet["request_id"], flush=True)
os.chdir(root)
os.environ["EFFIGY_CALLER"] = CALLER
os.execvp("effigy", ["effigy", "--host-run-request-id", packet["request_id"], "comparison"])
