#!/usr/bin/env python3
"""Disposable oxide#076 optimized-dependency quick-recipe measurement recipe.

Runs inside one heavy-admitted Effigy run. Not an Oxide product, not a reusable
launcher. One session on stable 1.97.1: one version call, then three arms
(A quick recipe, B + package."*".opt-level=1, C + opt-level=2). Per arm: one
cold `cargo test --no-run`, then three edit -> `cargo test --no-run` cycles;
after every build the reported test executables (profile.test only) run
directly and are timed. Caps: 12 measured Cargo calls + 2 version/setup calls,
the original 30-minute clock started by the submission recipe, 6 GiB owned
scratch, 20 GiB host reserve, 32 MiB raw logs. Never retries a call.
"""
import ctypes, errno, hashlib, json, os, pathlib, re, shlex, signal, subprocess, threading, time

ROOT = pathlib.Path(__file__).resolve().parent
RAW = ROOT / 'raw'
SRC = ROOT / 'src'
SCRATCH = ROOT / 't'
cfg = json.loads((ROOT / 'packet.json').read_text())
PIN = cfg['pin']
GIB = 1024 ** 3

lib = ctypes.CDLL('/usr/lib/libSystem.B.dylib')
lib.mach_continuous_time.restype = ctypes.c_uint64
class Scale(ctypes.Structure):
    _fields_ = [('numer', ctypes.c_uint32), ('denom', ctypes.c_uint32)]
scale = Scale(); assert lib.mach_timebase_info(ctypes.byref(scale)) == 0
def now(): return lib.mach_continuous_time() * scale.numer // scale.denom
START = cfg['start_ns']; DEADLINE = cfg['deadline_ns']
def offset(): return round((now() - START) / 1e9, 3)

def free_bytes(path='/private/tmp'):
    v = os.statvfs(path); return v.f_bavail * v.f_frsize

class Stop(Exception):
    pass

counts = {'measured': 0, 'setup': 0}
result = {'packet': cfg, 'calls': [], 'arms': {}, 'storage': {}, 'boundaries': [], 'outcome': 'running',
          'stop_reason': None, 'counts': counts, 'notes': []}
scratch_reference = {'free': free_bytes()}

def save():
    tmp = ROOT / 'result.json.tmp'
    tmp.write_text(json.dumps(result, indent=2) + '\n'); tmp.replace(ROOT / 'result.json')

def raw_bytes():
    return sum(p.stat().st_size for p in RAW.iterdir() if p.is_file())

def bounds():
    if now() >= DEADLINE - 10_000_000_000:
        raise Stop('original 30-minute deadline reached (10 s margin)')
    free = free_bytes()
    if free < 21 * GIB:
        raise Stop('host free space %d below 20 GiB reserve + 1 GiB margin' % free)
    # Conservative owned-scratch guard while a call runs: any host free-space
    # drop since the reference counts as owned.
    if scratch_reference['free'] - free > int(5.5 * GIB):
        raise Stop('host free space dropped more than 5.5 GiB since the scratch reference')
    if raw_bytes() > 32 * 1024 ** 2:
        raise Stop('32 MiB raw log cap reached')

def boundary(label):
    result['boundaries'].append({'label': label, 'offset_s': offset(), 'loadavg': os.getloadavg(),
                                 'host_free': free_bytes()})
    save()

def group_settled(pgid, limit=10.0):
    end = time.time() + limit
    while time.time() < end:
        try:
            os.killpg(pgid, 0)
        except OSError as error:
            if error.errno == errno.ESRCH:
                return 'ESRCH'
            if error.errno != errno.EPERM:
                return 'errno %d' % error.errno
        time.sleep(0.1)
    return None

def run(name, argv, env, kind, cwd=None, timeout=300, expect_zero=True):
    """kind: measured | setup | other. Counted calls increment before start."""
    bounds()
    if kind == 'measured':
        if counts['measured'] + 1 > cfg['measured_cargo_cap']:
            raise Stop('measured Cargo cap would be exceeded by ' + name)
        counts['measured'] += 1
    elif kind == 'setup':
        if counts['setup'] + 1 > cfg['setup_cargo_cap']:
            raise Stop('setup Cargo cap would be exceeded by ' + name)
        counts['setup'] += 1
    out = RAW / (name + '.stdout'); err = RAW / (name + '.stderr')
    rec = {'name': name, 'kind': kind, 'argv': [str(a) for a in argv], 'cwd': str(cwd or ROOT),
           'env': env, 'start_offset_s': offset(), 'loadavg_before': os.getloadavg(),
           'free_before': free_bytes(), 'counts_after_start': dict(counts)}
    result['calls'].append(rec); save()
    print(name, 'START', rec['start_offset_s'], flush=True)
    t0 = now(); stop = None; exited = {}
    with out.open('wb') as o, err.open('wb') as e:
        child = subprocess.Popen([str(a) for a in argv], cwd=str(cwd or ROOT), env=env,
                                 stdin=subprocess.DEVNULL, stdout=o, stderr=e,
                                 start_new_session=True)
        rec['pgid'] = child.pid
        done = threading.Event()
        def wait_child():
            child.wait(); exited['ns'] = now(); done.set()
        threading.Thread(target=wait_child, daemon=True).start()
        try:
            while not done.wait(0.05):
                if (now() - t0) / 1e9 > timeout:
                    raise Stop('%s exceeded its %d s call bound' % (name, timeout))
                bounds()
        except Stop as reason:
            stop = reason
            try:
                os.killpg(child.pid, signal.SIGTERM)
                if not done.wait(5):
                    os.killpg(child.pid, signal.SIGKILL)
            except ProcessLookupError:
                pass
            done.wait()
    rec['wall_s'] = round((exited['ns'] - t0) / 1e9, 4)
    rec['exit'] = child.returncode
    rec['group_settled'] = group_settled(child.pid)
    rec['end_offset_s'] = offset(); rec['loadavg_after'] = os.getloadavg()
    rec['free_after'] = free_bytes()
    rec['stdout'] = str(out.relative_to(ROOT)); rec['stderr'] = str(err.relative_to(ROOT))
    save()
    print(name, 'EXIT', child.returncode, rec['wall_s'], flush=True)
    if stop is not None:
        rec['stopped'] = str(stop); save(); raise stop
    if rec['group_settled'] != 'ESRCH':
        raise Stop('%s process group not positively empty: %s' % (name, rec['group_settled']))
    if expect_zero and child.returncode != 0:
        raise Stop('%s exited %s; see retained raw logs' % (name, child.returncode))
    return rec

def sha256(path):
    h = hashlib.sha256()
    with open(path, 'rb') as f:
        for block in iter(lambda: f.read(1 << 20), b''):
            h.update(block)
    return h.hexdigest()

WORKSPACE_UNITS = {'oxide_storage', 'oxide', 'oxide_storage_inventory', 'oxide_storage_lease',
                   'profile_backtrace'}

def unit_base(name):
    base = re.sub(r'^lib', '', name)
    base = re.split(r'-[0-9a-f]{16}', base)[0].split('.')[0]
    return base.replace('-', '_')

def tree_bytes(path):
    """Apparent and allocated bytes with hardlink union, split workspace/dependency.

    Attribution is by Cargo unit file/directory name below the profile
    directory (a name heuristic, not a unit graph).
    """
    seen = set(); totals = {'files': 0, 'apparent': 0, 'allocated': 0,
                           'workspace_allocated': 0, 'dependency_allocated': 0, 'other_allocated': 0}
    for base, dirs, names in os.walk(path):
        dirs.sort(); rel = pathlib.Path(base).relative_to(path).parts
        for n in sorted(names):
            st = os.lstat(os.path.join(base, n))
            if (st.st_dev, st.st_ino) in seen:
                continue
            seen.add((st.st_dev, st.st_ino))
            alloc = st.st_blocks * 512
            totals['files'] += 1; totals['apparent'] += st.st_size; totals['allocated'] += alloc
            parts = rel[1:]
            if not rel or rel[0] != 'oxide-quick':
                crate = None
            elif not parts or parts[0] == 'deps':
                crate = unit_base(n)
            elif parts[0] in ('incremental', '.fingerprint', 'build') and len(parts) >= 2:
                crate = unit_base(parts[1])
            else:
                crate = None
            if not crate:
                totals['other_allocated'] += alloc
            elif crate in WORKSPACE_UNITS:
                totals['workspace_allocated'] += alloc
            else:
                totals['dependency_allocated'] += alloc
    return totals

def du_k(path):
    out = subprocess.run(['/usr/bin/du', '-sk', str(path)], capture_output=True, text=True)
    return int(out.stdout.split()[0]) * 1024 if out.returncode == 0 else None

def owned_check(label):
    owned = du_k(ROOT)
    result['storage'].setdefault('owned_boundaries', []).append(
        {'label': label, 'offset_s': offset(), 'owned_allocated_bytes': owned, 'host_free': free_bytes(),
         'raw_log_bytes': raw_bytes()})
    save()
    if owned is None or owned > 6 * GIB:
        raise Stop('owned scratch %s exceeds 6 GiB or is unknown at %s' % (owned, label))

def rustc_unit(line):
    """Normalise one verbose rustc line: opt-level apart, all other settings."""
    cmd = line.strip()[len('Running `'):]
    cmd = cmd[:-1] if cmd.endswith('`') else cmd
    tokens = shlex.split(cmd)
    crate = tokens[tokens.index('--crate-name') + 1]
    source = next((t for t in tokens if t.endswith('.rs')), None)
    registry = cfg['host_cargo_home'] + '/registry/src/'
    workspace = source is not None and not source.startswith('/')
    opt = None; rest = []; i = 1
    while i < len(tokens):
        t = tokens[i]
        if t in ('--out-dir', '-L', '--extern'):
            i += 2; continue
        if t == '-C' and i + 1 < len(tokens):
            v = tokens[i + 1]
            if v.startswith('opt-level='):
                opt = v.split('=', 1)[1]; i += 2; continue
            if v.startswith(('metadata=', 'extra-filename=', 'incremental=')):
                rest.append('-C ' + v.split('=', 1)[0] + '=<varies>'); i += 2; continue
            rest.append('-C ' + v); i += 2; continue
        rest.append(t.replace(registry, '<registry>/').replace(str(ROOT), '<root>'))
        i += 1
    key = '%s|%s|%s|debuginfo=%s' % (crate, 'test' if '--test' in tokens else 'lib/bin',
                                     (source or '').replace(registry, '<registry>/'),
                                     ','.join(re.findall(r'-C debuginfo=(\S+)', cmd)) or 'absent')
    return {'key': key, 'crate': crate, 'workspace': workspace, 'opt_level': opt or '0 (absent)',
            'debuginfo': re.findall(r'-C debuginfo=(\S+)', cmd), 'incremental': '-C incremental=' in cmd,
            'build_script': (source or '').endswith('build.rs'),
            'proc_macro': 'proc-macro' in tokens,
            'settings_except_opt_level_sha256': hashlib.sha256(' '.join(rest).encode()).hexdigest()}

def parse_cargo(rec):
    stdout = (ROOT / rec['stdout']).read_text(errors='replace')
    stderr = (ROOT / rec['stderr']).read_text(errors='replace')
    artifacts = []; finished = None; executables = []
    for line in stdout.splitlines():
        if not line.startswith('{'):
            continue
        try:
            m = json.loads(line)
        except ValueError:
            continue
        if m.get('reason') == 'compiler-artifact':
            profile = m.get('profile') or {}
            artifacts.append({'package_id': m['package_id'], 'target': m['target']['name'],
                              'kinds': m['target']['kind'], 'fresh': m['fresh'], 'profile': profile,
                              'executable': m.get('executable')})
            if profile.get('test') is True and m.get('executable'):
                executables.append({'target': m['target']['name'], 'kinds': m['target']['kind'],
                                    'src_path': m['target']['src_path'].replace(str(SRC), '<src>'),
                                    'executable': m['executable']})
        elif m.get('reason') == 'build-finished':
            finished = m.get('success')
    units = [rustc_unit(l) for l in stderr.splitlines()
             if l.strip().startswith('Running `') and ' --crate-name ' in l]
    rec['finished_line'] = [l.strip() for l in stderr.splitlines() if l.strip().startswith('Finished ')]
    rec['executable_lines'] = [l.strip() for l in stderr.splitlines() if l.strip().startswith('Executable ')]
    rec['build_finished_success'] = finished
    rec['rebuilt'] = sorted({a['target'] + ':' + ','.join(a['kinds']) + ':' + str(a['profile'].get('test'))
                             for a in artifacts if not a['fresh']})
    rec['fresh_count'] = sum(1 for a in artifacts if a['fresh'])
    rec['rebuilt_count'] = sum(1 for a in artifacts if not a['fresh'])
    rec['artifact_profiles'] = [{'target': a['target'], 'kinds': a['kinds'], 'fresh': a['fresh'],
                                 'opt_level': a['profile'].get('opt_level'),
                                 'debuginfo': a['profile'].get('debuginfo'), 'test': a['profile'].get('test')}
                                for a in artifacts]
    rec['rustc_units'] = units
    rec['test_executables'] = executables
    save()
    return executables

TEST_LINE = re.compile(r'^test (\S+) \.\.\. (\S+)')

def run_tests(arm, label, executables):
    """Directly execute every reported test executable; default membership."""
    env = {'PATH': '/usr/bin:/bin', 'HOME': '/Users/tom', 'LANG': 'C',
           'OXIDE_TASK_SCRATCH_ROOT': str(SCRATCH), 'TMPDIR': str(SCRATCH), 'TMP': str(SCRATCH),
           'TEMP': str(SCRATCH)}
    cycle = {'arm': arm, 'label': label, 'binaries': []}
    result['arms'][arm].setdefault('test_cycles', []).append(cycle); save()
    if not executables:
        raise Stop('%s-%s reported no test executables' % (arm, label))
    total = 0.0
    for idx, exe in enumerate(executables):
        rec = run('%s-%s-test%d-%s' % (arm, label, idx, exe['target']), [exe['executable']], env, 'other',
                  cwd=SRC, timeout=180, expect_zero=False)
        text = (ROOT / rec['stdout']).read_text(errors='replace')
        members = [(m.group(1), m.group(2)) for m in map(TEST_LINE.match, text.splitlines()) if m]
        rec['test_results'] = [l for l in text.splitlines() if l.startswith('test result:')]
        rec['members'] = members
        rec['passed'] = sum(1 for _, s in members if s == 'ok')
        rec['ignored'] = sum(1 for _, s in members if s == 'ignored')
        rec['failed'] = sum(1 for _, s in members if s not in ('ok', 'ignored'))
        rec['executable_sha256'] = sha256(exe['executable'])
        total += rec['wall_s']
        cycle['binaries'].append({'name': rec['name'], 'target': exe['target'], 'kinds': exe['kinds'],
                                  'wall_s': rec['wall_s'], 'exit': rec['exit'], 'passed': rec['passed'],
                                  'ignored': rec['ignored'], 'failed': rec['failed']})
        save()
        if rec['exit'] != 0 or rec['failed']:
            result['outcome'] = 'no-go'
            raise Stop('%s failed (exit %s, %d failed)' % (rec['name'], rec['exit'], rec['failed']))
    cycle['harness_total_s'] = round(total, 4); save()

def base_env():
    tc_bin = cfg['toolchain']['bin']
    return {'HOME': '/Users/tom', 'USER': 'tom', 'LOGNAME': 'tom', 'LANG': 'C',
            'PATH': tc_bin + ':/usr/bin:/bin:/usr/sbin:/sbin',
            'RUSTC': tc_bin + '/rustc', 'RUSTDOC': tc_bin + '/rustdoc',
            'CARGO_HOME': cfg['host_cargo_home'], 'CARGO_TERM_COLOR': 'never',
            'OXIDE_TASK_SCRATCH_ROOT': str(SCRATCH), 'TMPDIR': str(SCRATCH), 'TMP': str(SCRATCH),
            'TEMP': str(SCRATCH),
            'CARGO_PROFILE_OXIDE_QUICK_INHERITS': 'dev',
            'CARGO_PROFILE_OXIDE_QUICK_DEBUG': 'line-tables-only'}

EDIT_FILE = SRC / 'src/dep_info.rs'
EDIT_REL = 'src/dep_info.rs'
BASE_VALUE = 32768
EDIT_LINE = 'pub const MAX_REFERENCES: usize = %d;\n'

def edit(n):
    text = EDIT_FILE.read_text()
    old = EDIT_LINE % (BASE_VALUE + n - 1); new = EDIT_LINE % (BASE_VALUE + n)
    assert text.count(old) == 1, 'edit %d anchor not unique' % n
    EDIT_FILE.write_text(text.replace(old, new))
    return {'edit': n, 'file': EDIT_REL, 'old': old.rstrip('\n'), 'new': new.rstrip('\n'),
            'sha256_after': sha256(EDIT_FILE), 'offset_s': offset()}

def git(*args, name):
    return run(name, ['/usr/bin/git', '-C', SRC] + list(args), {'PATH': '/usr/bin:/bin', 'HOME': '/Users/tom'},
               'other', timeout=60)

def measure_arm(arm):
    spec = cfg['arms'][arm]
    env = base_env(); env['CARGO_TARGET_DIR'] = str(ROOT / ('target-' + arm))
    pre = [cfg['toolchain']['bin'] + '/cargo'] + spec.get('global_args', [])
    common = ['--workspace', '--all-targets', '--locked', '--offline', '--profile', 'oxide-quick', '--no-run',
              '-v', '--message-format=json-render-diagnostics']
    arm_rec = {'spec': spec, 'env': env, 'steps': [], 'builds': []}; result['arms'][arm] = arm_rec
    boundary(arm + ' start')
    git('reset', '--quiet', '--hard', PIN, name=arm + '-reset')
    git('clean', '-fdq', name=arm + '-clean')
    status = git('status', '--porcelain', name=arm + '-status')
    assert (ROOT / status['stdout']).read_text() == '', 'source not clean after reset'
    arm_rec['baseline_edit_file_sha256'] = sha256(EDIT_FILE)
    for n in (0, 1, 2, 3):
        label = 'cold' if n == 0 else 'e%d' % n
        if n:
            arm_rec.setdefault('edits', []).append(edit(n)); save()
        rec = run('%s-%s-build' % (arm, label), pre + ['test'] + common, env, 'measured', cwd=SRC,
                  timeout=600 if n == 0 else 300, expect_zero=False)
        executables = parse_cargo(rec); arm_rec['steps'].append(rec['name'])
        arm_rec['builds'].append({'label': label, 'name': rec['name'], 'wall_s': rec['wall_s'], 'exit': rec['exit'],
                                  'fresh_count': rec['fresh_count'], 'rebuilt_count': rec['rebuilt_count'],
                                  'rebuilt': rec['rebuilt']})
        save()
        if rec['exit'] != 0:
            raise Stop('%s exited %s' % (rec['name'], rec['exit']))
        run_tests(arm, label, executables)
        boundary('%s after %s' % (arm, label))
        if n == 0:
            arm_rec['storage_after_cold'] = tree_bytes(ROOT / ('target-' + arm)); save()
            owned_check(arm + ' after cold')
    arm_rec['storage_final'] = tree_bytes(ROOT / ('target-' + arm)); save()
    owned_check(arm + ' final')

def compare_settings():
    """Prove only dependency opt-level differs between arms (cold builds)."""
    cold = {}
    for arm in cfg['arm_order']:
        rec = next((c for c in result['calls'] if c['name'] == arm + '-cold-build'), None)
        if rec is None or 'rustc_units' not in rec:
            return
        cold[arm] = {u['key']: u for u in rec['rustc_units']}
        if len(cold[arm]) != len(rec['rustc_units']):
            result['notes'].append('%s cold build has duplicate unit keys' % arm)
    ref = cold[cfg['arm_order'][0]]
    out = {}
    for arm, units in cold.items():
        same_keys = set(units) == set(ref)
        diff_settings = sorted(k for k in units if k in ref and units[k]['settings_except_opt_level_sha256']
                               != ref[k]['settings_except_opt_level_sha256'])
        opt = {}
        for u in units.values():
            cls = 'workspace' if u['workspace'] else ('dependency build script' if u['build_script'] else
                                                      ('dependency proc-macro' if u['proc_macro'] else 'dependency'))
            opt.setdefault(cls, {}).setdefault(u['opt_level'], []).append(u['crate'])
        out[arm] = {'unit_count': len(units), 'same_unit_keys_as_first_arm': same_keys,
                    'units_with_other_setting_differences': diff_settings,
                    'opt_level_by_class': {c: {k: sorted(v) for k, v in m.items()} for c, m in opt.items()}}
    result['settings_comparison'] = out; save()

def main():
    result['session'] = {'start_ns': START, 'deadline_ns': DEADLINE, 'clock': cfg['clock'],
                         'admitted_offset_s': offset(), 'admitted_utc': time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime())}
    SCRATCH.mkdir(mode=0o700, exist_ok=True)
    assert os.path.realpath(SCRATCH) == str(SCRATCH), 'scratch root is not canonical'
    boundary('admitted')
    try:
        b = cfg['toolchain']['bin']
        venv = {'PATH': b + ':/usr/bin:/bin', 'HOME': '/Users/tom', 'CARGO_HOME': cfg['host_cargo_home']}
        run('cargo-version', [b + '/cargo', '-vV'], venv, 'setup', timeout=60)
        run('rustc-version', [b + '/rustc', '-vV'], venv, 'other', timeout=60)
        libdir = pathlib.Path(b).parent / 'lib'
        result['toolchain_identity'] = {
            'cargo_sha256': sha256(b + '/cargo'), 'rustc_sha256': sha256(b + '/rustc'),
            'rustc_driver': {d.name: sha256(d) for d in sorted(libdir.glob('librustc_driver-*.dylib'))}}
        save()
        run('clone', ['/usr/bin/git', 'clone', '--quiet', '--no-hardlinks', '--no-checkout',
                      cfg['source_repository'], SRC], {'PATH': '/usr/bin:/bin', 'HOME': '/Users/tom'}, 'other',
            timeout=120)
        git('checkout', '--quiet', '--detach', PIN, name='checkout')
        head = git('rev-parse', 'HEAD', 'HEAD^{tree}', name='rev-parse')
        result['source'] = {'head_and_tree': (ROOT / head['stdout']).read_text().split(),
                            'lock_sha256': sha256(SRC / 'Cargo.lock')}
        assert result['source']['head_and_tree'][0] == PIN, 'clone HEAD is not the pin'
        scratch_reference['free'] = free_bytes(); result['storage']['scratch_reference_free'] = scratch_reference['free']
        owned_check('after setup')
        for arm in cfg['arm_order']:
            measure_arm(arm)
        result['outcome'] = 'complete'
    except Stop as reason:
        result['stop_reason'] = str(reason)
        if result['outcome'] == 'running':
            result['outcome'] = 'partial'
    except Exception as error:  # preserve partial result on harness failure
        result['stop_reason'] = 'harness error: %r' % (error,)
        result['outcome'] = 'partial'
    try:
        compare_settings()
    except Exception as error:
        result['notes'].append('settings comparison failed: %r' % (error,))
    boundary('end')
    try:
        owned_check('end')
    except Stop as reason:
        result['notes'].append(str(reason))
    result['session']['end_offset_s'] = offset()
    result['raw_log_bytes'] = raw_bytes()
    save()
    print('OUTCOME', result['outcome'], result['stop_reason'], 'counts', counts, flush=True)
    return 0 if result['outcome'] == 'complete' else 2

if __name__ == '__main__':
    raise SystemExit(main())
