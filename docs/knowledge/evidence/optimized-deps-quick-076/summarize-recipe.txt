#!/usr/bin/env python3
"""Derive summary.json from the retained oxide#076 result.json. No measurement."""
import json, statistics, sys, pathlib

root = pathlib.Path(sys.argv[1])
r = json.loads((root / 'result.json').read_text())
MIB = 1024 ** 2
calls = {c['name']: c for c in r['calls']}
out = {'outcome': r['outcome'], 'stop_reason': r['stop_reason'], 'counts': r['counts'],
       'pin': r['packet']['pin'], 'toolchain': r['packet']['toolchain']['name'],
       'utc_start': r['packet']['utc_start'], 'session': r['session'],
       'raw_log_bytes': r.get('raw_log_bytes'), 'arms': {}, 'boundaries': r['boundaries'],
       'owned_boundaries': r['storage'].get('owned_boundaries'),
       'settings_comparison': r.get('settings_comparison'), 'notes': r['notes']}
for arm, a in r['arms'].items():
    builds = a.get('builds', []); cycles = a.get('test_cycles', [])
    warm = [b['wall_s'] for b in builds if b['label'] != 'cold']
    warm_tests = [c.get('harness_total_s') for c in cycles if c['label'] != 'cold' and 'harness_total_s' in c]
    membership = {}
    for c in cycles:
        for b in c['binaries']:
            membership.setdefault(b['target'] + ':' + ','.join(b['kinds']), set()).add(
                (b['passed'], b['ignored'], b['failed']))
    out['arms'][arm] = {
        'what': a['spec']['what'],
        'builds': [{k: b[k] for k in ('label', 'wall_s', 'exit', 'fresh_count', 'rebuilt_count')} for b in builds],
        'cold_build_s': next((b['wall_s'] for b in builds if b['label'] == 'cold'), None),
        'warm_build_mean_s': round(statistics.mean(warm), 4) if warm else None,
        'test_cycles': [{'label': c['label'], 'harness_total_s': c.get('harness_total_s'),
                         'per_binary_s': {b['target'] + ':' + ','.join(b['kinds']): b['wall_s'] for b in c['binaries']}}
                        for c in cycles],
        'warm_harness_mean_s': round(statistics.mean(warm_tests), 4) if warm_tests else None,
        'membership_passed_ignored_failed': {k: sorted(v) for k, v in membership.items()},
        'target_after_cold_mib': round(a['storage_after_cold']['allocated'] / MIB, 1) if 'storage_after_cold' in a else None,
        'target_final': {k: (round(v / MIB, 1) if k != 'files' else v) for k, v in a['storage_final'].items()}
        if 'storage_final' in a else None,
        'edits': [{k: e[k] for k in ('edit', 'file', 'old', 'new')} for e in a.get('edits', [])],
    }
(root / 'summary.json').write_text(json.dumps(out, indent=2) + '\n')
print(json.dumps({k: {kk: vv for kk, vv in v.items() if kk in ('cold_build_s', 'warm_build_mean_s',
                                                              'warm_harness_mean_s', 'target_after_cold_mib')}
                  for k, v in out['arms'].items()}, indent=1))
print(out['outcome'], out['stop_reason'], out['counts'], out['session'].get('end_offset_s'))
